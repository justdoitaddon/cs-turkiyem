/* ===== Teşhis toplayıcı (window.__czDiag) =====
   "Hata Bildir" formuna otomatik teknik bilgi eklemek için sayfa açılır
   açılmaz console.error/warn + global JS hatalarını + yakalanmamış promise
   reddini halka-tampona (ring buffer) yazar. Hiçbir şey otomatik gönderilmez;
   yalnızca kullanıcı bildiri gönderince snapshot'la birlikte yollanır.
   Player çoğunlukla native ArtPlayer (HDFC/Byse/Fkr/Sibnet pass-through/Dzen/
   np...) olduğundan hls.js hata logları ('[hdfc-hls]' vb.) ve <video> error
   state'i de buradan yakalanır. Saf iframe embed'lerin (ok.ru, dzen iframe...)
   konsolu cross-origin olduğu için okunamaz — bu durumda en azından aktif
   sunucu + sayfa-seviyesi hatalar kaydedilir. */
(function () {
    if (window.__czDiag) return;
    var MAX = 45, MSG_CAP = 400;
    var t0 = Date.now();
    var buf = [];
    function push(level, parts) {
        try {
            var msg = parts.map(function (p) {
                if (p instanceof Error) return (p.name || 'Error') + ': ' + (p.message || '') + (p.stack ? ' | ' + String(p.stack).split('\n').slice(0, 3).join(' ') : '');
                if (typeof p === 'object') { try { return JSON.stringify(p); } catch (e) { return String(p); } }
                return String(p);
            }).join(' ');
            if (msg.length > MSG_CAP) msg = msg.slice(0, MSG_CAP) + '…';
            buf.push({ ms: Date.now() - t0, l: level, m: msg });
            if (buf.length > MAX) buf.shift();
        } catch (e) {}
    }
    window.__czDiag = { buf: buf, push: push, since: t0 };

    // console.error / console.warn'u sar (orijinali koru).
    ['error', 'warn'].forEach(function (lvl) {
        var orig = console[lvl];
        console[lvl] = function () {
            try { push(lvl, [].slice.call(arguments)); } catch (e) {}
            if (typeof orig === 'function') return orig.apply(console, arguments);
        };
    });

    // Global JS hataları + kaynak yükleme hataları (capture: true → <script>/<img>
    // gibi kaynaklar da yakalanır; cross-origin script "Script error." verir).
    window.addEventListener('error', function (e) {
        try {
            if (e && e.target && e.target !== window && (e.target.src || e.target.href)) {
                push('reserr', ['kaynak yüklenemedi:', e.target.tagName, e.target.src || e.target.href]);
            } else {
                push('jserr', [(e && e.message) || 'error', (e && e.filename) ? ('@' + e.filename + ':' + e.lineno + ':' + e.colno) : '']);
            }
        } catch (er) {}
    }, true);

    window.addEventListener('unhandledrejection', function (e) {
        try {
            var r = e && e.reason;
            push('reject', [r instanceof Error ? r : ('promise reddi: ' + (typeof r === 'object' ? JSON.stringify(r) : String(r)))]);
        } catch (er) {}
    });

    // Ağ durumu değişimi.
    window.addEventListener('offline', function () { push('net', ['çevrimdışı (offline) olundu']); });
    window.addEventListener('online',  function () { push('net', ['çevrimiçi (online) olundu']); });
})();

    function insertAtCursor(el, text) {
        var start = (el.selectionStart != null) ? el.selectionStart : el.value.length;
        var end   = (el.selectionEnd   != null) ? el.selectionEnd   : el.value.length;
        var v = el.value;
        el.value = v.slice(0, start) + text + v.slice(end);
        var pos = start + text.length;
        el.selectionStart = el.selectionEnd = pos;
    }

    // Renk skalasi - rate butonu ve yorum rozeti ortak kullanir.
    function scoreColorClass(s) {
        s = s | 0;
        if (s >= 8) return 'score-good';
        if (s >= 6) return 'score-mid';
        if (s >= 1) return 'score-bad';
        return '';
    }

    // Bir yorum kartindaki ★ X/10 rozetini guncelle (puan 0 ise gizle).
    function applyScoreBadge(node, score) {
        var badge = node.querySelector('.cz-c-score');
        if (!badge) return;
        var s = score | 0;
        badge.classList.remove('score-good', 'score-mid', 'score-bad');
        if (s > 0) {
            var numEl = badge.querySelector('.cz-c-score-num');
            if (numEl) numEl.textContent = s;
            badge.hidden = false;
            var cc = scoreColorClass(s);
            if (cc) badge.classList.add(cc);
        } else {
            badge.hidden = true;
        }
    }

    function fmtDate(iso) {
        if (!iso) return '';
        var d = new Date(iso.replace(' ', 'T') + 'Z');
        if (isNaN(d.getTime())) return iso;
        var pad = function (n) { return n < 10 ? '0' + n : '' + n; };
        return pad(d.getDate()) + '/' + pad(d.getMonth() + 1) + '/' + d.getFullYear()
             + ' ' + pad(d.getHours()) + ':' + pad(d.getMinutes()) + ':' + pad(d.getSeconds());
    }

    function escapeHtml(s) {
        return String(s).replace(/[&<>"']/g, function (c) {
            return ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'})[c];
        });
    }

    // [spoiler]...[/spoiler] -> tıklayınca açılan kutu. İç metin escape edilir.
    function renderBody(s) {
        var esc = escapeHtml(s);
        return esc.replace(/\[spoiler\]([\s\S]*?)\[\/spoiler\]/gi, function (_, inner) {
            return '<span class="cz-c-spoiler-tag" tabindex="0" role="button" aria-pressed="false" title="Spoileri göster">' +
                       '<span class="cz-c-spoiler-tag-label">SPOILER</span>' +
                       '<span class="cz-c-spoiler-tag-text">' + inner + '</span>' +
                   '</span>';
        });
    }
    function bindSpoilers(node) {
        node.querySelectorAll('.cz-c-spoiler-tag').forEach(function (s) {
            if (s.dataset.czSpoilerBound === '1') return;
            s.dataset.czSpoilerBound = '1';
            function reveal() {
                s.classList.add('is-open');
                s.setAttribute('aria-pressed', 'true');
                s.removeAttribute('title');
            }
            s.addEventListener('click', reveal);
            s.addEventListener('keydown', function (e) {
                if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); reveal(); }
            });
        });
    }

    // ---- Cevap (reply) compose kutusu ----
    // rootNode: cevabin baglanacagi kok .cz-comment (top-level)
    // targetNode: "Yanitla"ya basilan dugum (cevap ardina yerlestirme icin)
    // targetId: server'a parent_id olarak gidecek yorum id'si (tiklanan yorum)
    // mentionName: cevaba cevap veriliyorsa @isim on-dolgusu
    function openReplyCompose(rootNode, targetNode, targetId, mentionName) {
        if (!rootNode) return;
        var repliesWrap = rootNode.querySelector('.cz-c-replies');
        if (!repliesWrap) {
            repliesWrap = document.createElement('div');
            repliesWrap.className = 'cz-c-replies';
            rootNode.appendChild(repliesWrap);
        }
        // Tek aktif compose: kok altinda baska compose varsa kaldir.
        var existing = repliesWrap.querySelector('.cz-c-reply-compose');
        if (existing) existing.remove();

        var box = document.createElement('div');
        box.className = 'cz-c-reply-compose';
        box.innerHTML =
            '<textarea class="cz-c-reply-ta" maxlength="1000" rows="2" placeholder="Yanıtını yaz…"></textarea>' +
            '<div class="cz-c-reply-actions">' +
                '<button type="button" class="cz-c-reply-cancel">İptal</button>' +
                '<button type="button" class="cz-c-reply-send"><i class="fas fa-reply"></i> Yanıtla</button>' +
            '</div>';

        // Yerlestir: hedef bir cevapsa onun hemen ardina, degilse listenin sonuna.
        if (targetNode && targetNode.classList.contains('is-reply') && targetNode.parentNode === repliesWrap) {
            targetNode.insertAdjacentElement('afterend', box);
        } else {
            repliesWrap.appendChild(box);
        }

        var rta = box.querySelector('.cz-c-reply-ta');
        if (mentionName) rta.value = '@' + mentionName + ' ';
        try { rta.focus(); rta.setSelectionRange(rta.value.length, rta.value.length); } catch (e) {}

        box.querySelector('.cz-c-reply-cancel').addEventListener('click', function () { box.remove(); });
        var sendBtn = box.querySelector('.cz-c-reply-send');
        sendBtn.addEventListener('click', function () { postReply(rootNode, targetId, rta, sendBtn, box); });
        rta.addEventListener('keydown', function (e) {
            if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') sendBtn.click();
        });
    }

    function replyErrText(err) {
        return ({
            'rate_limited':    'Çok hızlı yazıyorsun, biraz bekle.',
            'empty_body':      'Yanıt boş olamaz.',
            'too_long':        'Yanıt çok uzun.',

    function postReply(rootNode, parentId, rta, sendBtn, box) {
        var body = (rta.value || '').trim();
        if (!body) { rta.focus(); return; }
        if (body.length > 1000) { showToast('Yanıt çok uzun (max 1000 karakter).', 'info'); return; }
        var gName = nameInp ? ((nameInp.value || '').trim() || (localStorage.getItem('cz-c-name') || '')) : '';
        sendBtn.disabled = true;
        var payload = {
            anime_id:     animeId,
            episode_id:   episodeId,
            parent_id:    parentId,
            body:         body,
            author_name:  gName,
            bg_character: 'none'
        };
        fetch('/api/comments/', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
            body: JSON.stringify(payload)
        })
            .then(function (r) { return r.json().then(function (j) { return { status: r.status, j: j }; }); })
            .then(function (res) {
                sendBtn.disabled = false;
                if (!res.j.ok) { showToast(replyErrText(res.j.error || ('http_' + res.status)), 'info'); return; }
                box.remove();
                loadedComments.push(res.j.comment);
                renderOne(res.j.comment, true);
                applyListLimit();
                showToast('Yanıtın yayınlandı.', 'success');
            })
            .catch(function () {
                sendBtn.disabled = false;
                showToast('Bağlantı hatası, tekrar dene.', 'info');
            });
    }

    function setStatus(msg, kind) {
        statusEl.textContent = msg || '';
        statusEl.classList.remove('is-error', 'is-ok');
        if (kind) statusEl.classList.add('is-' + kind);
        if (msg) setTimeout(function () { if (statusEl.textContent === msg) { statusEl.textContent = ''; statusEl.classList.remove('is-error', 'is-ok'); } }, 4000);
    }

    // Yuklenmis yorumlarin canonical kopyasi - siralama bar'i bu diziyi mutate edip render eder.
    var loadedComments = [];

    function renderListFromState() {
        listEl.innerHTML = '';
        if (!loadedComments.length) {
            listEl.innerHTML = '<div class="cz-c-empty">Henüz yorum yok. İlk yorumu sen yaz!</div>';
            return;
        }
        // Once kok yorumlar (mevcut siralamayla), sonra cevaplar (kronolojik).
        // Boylece bir cevap render edilirken kok dugumu zaten DOM'da bulunur.
        var parents = loadedComments.filter(function (c) { return !c.parent_id; });
        var replies = loadedComments.filter(function (c) { return c.parent_id; });
        parents.forEach(function (c) { renderOne(c, false); });
        replies.sort(function (a, b) {
            var av = String(a.created_at || ''), bv = String(b.created_at || '');
            if (av < bv) return -1;
            if (av > bv) return  1;
            return (a.id | 0) - (b.id | 0);
        });
        replies.forEach(function (c) { renderOne(c, false); });
        applyListLimit();
    }

    function loadComments() {
        listEl.innerHTML = '<div class="cz-c-empty">Yorumlar yükleniyor…</div>';
        var url = '/api/comments/?anime_id=' + animeId;
        if (episodeId) url += '&episode_id=' + episodeId;
        fetch(url, { headers: { 'Accept': 'application/json' } })
            .then(function (r) { return r.json(); })
            .then(function (j) {
                loadedComments = (j && j.ok && Array.isArray(j.comments)) ? j.comments : [];
                renderListFromState();
            })
            .catch(function () {
                listEl.innerHTML = '<div class="cz-c-empty">Yorumlar yüklenemedi.</div>';
            });
    }

    // Siralama: kriter = recent (created_at) | score (user_score) | popular (like_count); yon = asc/desc
    function applySort() {
        var dirSel  = root.querySelector('.cz-c-ctrl-dir');
        var sortSel = root.querySelector('.cz-c-ctrl-sort');
        var dir   = dirSel  ? dirSel.value  : 'desc';
        var field = sortSel ? sortSel.value : 'recent';
        var mul = (dir === 'asc') ? 1 : -1;
        loadedComments.sort(function (a, b) {
            var av, bv;
            if (field === 'score')        { av = (a.user_score | 0); bv = (b.user_score | 0); }
            else if (field === 'popular') { av = (a.like_count | 0); bv = (b.like_count | 0); }
            else /* recent */              { av = String(a.created_at || ''); bv = String(b.created_at || ''); }
            if (av < bv) return -1 * mul;
            if (av > bv) return  1 * mul;
            return ((b.id | 0) - (a.id | 0)); // esitlikte yeni olan basa
        });
        renderListFromState();
    }

    function resetSort() {
        var dirSel  = root.querySelector('.cz-c-ctrl-dir');
        var sortSel = root.querySelector('.cz-c-ctrl-sort');
        if (dirSel)  dirSel.value  = 'desc';
        if (sortSel) sortSel.value = 'recent';
        applySort();
    }

    var ctrlApply = root.querySelector('.cz-c-ctrl-apply');
    var ctrlReset = root.querySelector('.cz-c-ctrl-reset');
    if (ctrlApply) ctrlApply.addEventListener('click', applySort);
    if (ctrlReset) ctrlReset.addEventListener('click', resetSort);

    function postComment() {
        var body = (ta.value || '').trim();
        // GIF/foto eklendiyse body bos olabilir
        if (!body && !pendingMedia) { setStatus('Yorum boş olamaz.', 'error'); ta.focus(); return; }
        if (body.length > 1000) { setStatus('Yorum çok uzun (max 1000 karakter).', 'error'); return; }
        var name = nameInp ? ((nameInp.value || '').trim() || 'Misafir') : '';
        if (nameInp && name) localStorage.setItem('cz-c-name', name);

        btn.disabled = true;
        setStatus('Gönderiliyor…');

        var payload = {
            anime_id:     animeId,
            episode_id:   episodeId,
            body:         body,
            author_name:  name,
            bg_character: bgSel.value
        };
        if (pendingMedia) {
            payload.media_type = pendingMedia.type;
            payload.media_url  = pendingMedia.url;
            payload.media_meta = {
                w: pendingMedia.w | 0,
                h: pendingMedia.h | 0,
                preview:  pendingMedia.preview || '',
                tenor_id: pendingMedia.tenor_id || ''
            };
        }

        fetch('/api/comments/', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
            body: JSON.stringify(payload)
        })
            .then(function (r) { return r.json().then(function (j) { return { status: r.status, j: j }; }); })
            .then(function (res) {
                btn.disabled = false;
                if (!res.j.ok) {
                    var err = res.j.error || ('http_' + res.status);
                    var human = ({
                        'rate_limited':    'Çok hızlı yazıyorsun, biraz bekle.',
                        'empty_body':      'Yorum boş olamaz.',
                        'too_long':        'Yorum çok uzun.',
                        'anime_not_found': 'Bu içerik bulunamadı.',
                        'comment_banned':  'Yorumlardan engellendin, şu an yorum yapamazsın.'
                    })[err] || ('Hata: ' + err);
                    setStatus(human, 'error');
                    return;
                }
                ta.value = '';
                setPendingMedia(null);
                setStatus('');
                // Empty state varsa kaldır
                var empty = listEl.querySelector('.cz-c-empty');
                if (empty) empty.remove();
                loadedComments.unshift(res.j.comment);
                renderOne(res.j.comment, true);
                applyListLimit();
                showToast('Yorumun yayınlandı.', 'success');
            })
            .catch(function () {
                btn.disabled = false;
                setStatus('Bağlantı hatası, tekrar dene.', 'error');
            });
    }

    function doLike(id, node) {
        var likeBtn = node.querySelector('.cz-c-like');
        var countEl = node.querySelector('.cz-c-like-count');
        likeBtn.disabled = true;
        fetch('/api/comments/' + id + '/like/', { method: 'POST', headers: { 'Accept': 'application/json' } })
            .then(function (r) { return r.json(); })
            .then(function (j) {
                likeBtn.disabled = false;

            var v = ta.value;
            var start = ta.selectionStart != null ? ta.selectionStart : v.length;
            var end   = ta.selectionEnd   != null ? ta.selectionEnd   : v.length;
            var hasSel = end > start;

            ta.focus();

            if (hasSel) {
                var sel = v.slice(start, end);
                ta.value = v.slice(0, start) + open + sel + close + v.slice(end);
                var s1 = start + open.length;
                ta.setSelectionRange(s1, s1 + sel.length);
                return;
            }

            var trimmed = v.trim();
            if (trimmed === '') {
                // Boş textarea: imlece boş spoiler bırak
                ta.value = open + close;
                ta.setSelectionRange(open.length, open.length);
                return;
            }

            // Toggle: tüm yazı zaten [spoiler]...[/spoiler] içindeyse, etiketleri çıkar
            var m = /^\s*\[spoiler\]([\s\S]*)\[\/spoiler\]\s*$/i.exec(v);
            if (m) {
                ta.value = m[1];
                ta.setSelectionRange(0, ta.value.length);
                return;
            }

            // Tüm yazıyı sar
            ta.value = open + v + close;
            ta.setSelectionRange(open.length, open.length + v.length);

function czInitBgDropdown(root, nativeSel) {
    var dd = root.querySelector('.cz-c-bg-dd');
    if (!dd || dd.dataset.czInited === '1') return;
    dd.dataset.czInited = '1';

    var trigger    = dd.querySelector('.cz-c-bg-trigger');
    var triggerLbl = dd.querySelector('.cz-c-bg-trigger-label');
    var panel      = dd.querySelector('.cz-c-bg-panel');
    var search     = dd.querySelector('.cz-c-bg-search');
    var optsWrap   = dd.querySelector('.cz-c-bg-options');
    var emptyEl    = dd.querySelector('.cz-c-bg-empty');
    var options    = Array.prototype.slice.call(dd.querySelectorAll('.cz-c-bg-option'));
    var groups     = Array.prototype.slice.call(dd.querySelectorAll('.cz-c-bg-group'));

    function open() {
        if (dd.classList.contains('is-open')) return;
        dd.classList.add('is-open');
        trigger.setAttribute('aria-expanded', 'true');
        // Aramayı sıfırla, focus'la
        search.value = '';
        applyFilter('');
        setTimeout(function () { try { search.focus(); } catch (e) {} }, 0);
        // Seçili olanı görüşe getir
        var sel = optsWrap.querySelector('.cz-c-bg-option[aria-selected="true"]');
        if (sel) {
            try { sel.scrollIntoView({ block: 'nearest' }); } catch (e) {}
        }
    }
    function close() {
        if (!dd.classList.contains('is-open')) return;
        dd.classList.remove('is-open');
        trigger.setAttribute('aria-expanded', 'false');
    }

    trigger.addEventListener('click', function (e) {
        e.stopPropagation();
        if (dd.classList.contains('is-open')) close(); else open();
    });

    // Dışarı tıklama -> kapat
    document.addEventListener('click', function (ev) {
        if (!dd.classList.contains('is-open')) return;
        if (dd.contains(ev.target)) return;
        close();
    });

    // ESC -> kapat
    dd.addEventListener('keydown', function (ev) {
        if (ev.key === 'Escape') { close(); trigger.focus(); }
    });

    // Seçim
    options.forEach(function (opt) {
        opt.addEventListener('click', function () {
            var val = opt.getAttribute('data-value');
            var lbl = opt.getAttribute('data-label') || opt.textContent.trim();
            options.forEach(function (o) { o.removeAttribute('aria-selected'); });
            opt.setAttribute('aria-selected', 'true');
            dd.setAttribute('data-value', val);
            triggerLbl.textContent = lbl;
            // Native select'i güncelle + change event (mevcut kod bunu dinliyor)
            if (nativeSel.value !== val) {
                nativeSel.value = val;
                nativeSel.dispatchEvent(new Event('change', { bubbles: true }));
            }
            close();
        });
    });

    function norm(s) {
        return (s || '').toString().toLowerCase()
            .replace(/ı/g, 'i').replace(/İ/g, 'i')
            .replace(/ş/g, 's').replace(/ğ/g, 'g')
            .replace(/ü/g, 'u').replace(/ö/g, 'o').replace(/ç/g, 'c');
    }

    function applyFilter(q) {
        q = norm(q.trim());
        var anyVisible = false;
        groups.forEach(function (g) {
            var groupName = g.getAttribute('data-group') || '';
            var groupMatch = q && norm(groupName).indexOf(q) !== -1;
            var visibleInGroup = 0;
            var optsInGroup = g.querySelectorAll('.cz-c-bg-option');
            optsInGroup.forEach(function (opt) {
                var label = opt.getAttribute('data-label') || '';
                var match = !q || groupMatch || norm(label).indexOf(q) !== -1;
                opt.hidden = !match;
                if (match) visibleInGroup++;
            });
            g.hidden = visibleInGroup === 0;
            if (visibleInGroup > 0) anyVisible = true;
        });
        if (emptyEl) emptyEl.hidden = anyVisible;
    }

    search.addEventListener('input', function () { applyFilter(search.value); });
    // Tıklamada panel kapanmasın
    panel.addEventListener('click', function (e) { e.stopPropagation(); });
}

    var _serverAttempted = {};

    function _serverKey(s) {
        if (!s) return '';
        return String(s.src || s.streamUrl || s.resolveUrl || s.metaUrl || (s.type + ':' + (s.embed_id || s.id || '')));
    }
    function _markAttempted(s) {

        if (k) _serverAttempted[k] = true;
    }
    function _isAttempted(s) {
        return !!_serverAttempted[_serverKey(s)];
    }
    // Aktif servers listesinde startIdx'ten sonraki ilk denenmemis index'i dondurur.
    function _findNextUntriedInCurrent(startIdx) {
        if (!servers || !servers.length) return -1;
        for (var i = Math.max(0, startIdx + 1); i < servers.length; i++) {
            if (!_isAttempted(servers[i])) return i;
        }
        return -1;
    }
    // Diger dil havuzunda (dub<->sub) denenmemis bir embed varsa servers'i o havuza
    // gecirir ve idx'ini doner. {lang, idx} | null.
    function _findUntriedInOtherLang() {
        if (!serversByLang) return null;
        var langs = Object.keys(serversByLang);
        for (var li = 0; li < langs.length; li++) {
            var lg = langs[li];
            if (lg === currentLang) continue;
            var arr = serversByLang[lg];
            if (!arr || !arr.length) continue;
            for (var i = 0; i < arr.length; i++) {
                if (!_isAttempted(arr[i])) return { lang: lg, idx: i };
            }
        }
        return null;
    }
    // Sonraki denenmemis embed'i mount eder; bulamazsa false doner.
    function _tryNextUntried(opts) {
        var next = _findNextUntriedInCurrent(current);
        if (next >= 0) {
            renderServer(next, opts || {});
            return true;
        }
        var other = _findUntriedInOtherLang();
        if (other) {
            servers = serversByLang[other.lang];
            currentLang = other.lang;
            current = 0;
            renderServer(other.idx, opts || {});
            return true;
        }
        return false;

            iframe.setAttribute('allowfullscreen', '');
            iframe.setAttribute('scrolling', 'no');
            iframe.setAttribute('frameborder', '0');
            wrap.insertBefore(iframe, wrap.firstChild);

            // Iframe load watchdog: cross-origin oldugu icin icerideki "Video
            // yuklenemedi" UI'sini okuyamayiz.
            //
            // ⚠️ 'load' event'i TEK BASINA canlilik kaniti DEGIL: tarayici KENDI
            // hata sayfasini (ERR_NAME_NOT_RESOLVED, 502/503, ERR_CONNECTION_*)
            // iframe'e bastiginda da 'load' firlar. Eski kod bunu "yuklendi"
            // sayip watchdog'u iptal ediyordu; 2026-08-17 sibnet cokusunde
            // (HAProxy 503 + bazi ISP resolver'larinda cozulememe) failover HIC
            // tetiklenmedi ve kullanici calisan alternatif sunucu dururken
            // "video.sibnet.ru ana makinesinin sunucu IP adresi bulunamadi"
            // ekranina kilitlendi.
            //
            // Cozum: 'load'a guvenmek yerine BAGIMSIZ erisilebilirlik probe'u.
            // s.src bizim same-origin /oynat/?t=<token> adresimiz; 302 ile
            // saglayiciya gidiyor. redirect:'follow' + mode:'no-cors' ile fetch
            // edince redirect zinciri kullanicinin kendi baglantisi uzerinden
            // yurur — saglayici cozulemiyor/erisilemiyorsa promise REJECT olur.
            // (Saglayici host'u HTML'e yazmadan probe edebilmemizin tek yolu bu;
            // view-source'da provider adi gozukmemeli — bkz. NATIVE_TYPES base64.)
            // Opak cevap status okunamaz, yani 503'u ayirt edemeyiz; o vaka icin
            // 12sn'lik no-load watchdog'u ikinci katman olarak duruyor.
            var _iframeLoaded = false;
            var _iframeDead   = false;

            function _iframeFail(why) {
                if (_iframeDead) return;               // tek atis
                if (!iframe.isConnected) return;       // baska sunucuya gecilmis, bu mount bayat
                _iframeDead = true;
                if (window.console) console.warn('[iframe] kaynak erisilemez (' + why + ') → sonraki sunucuya geciliyor');
                if (_iframeWatchTimer) { clearTimeout(_iframeWatchTimer); _iframeWatchTimer = null; }
                _handleServerFailure();
            }

            iframe.addEventListener('load', function () {
                _iframeLoaded = true;
                // NOT: watchdog'u burada IPTAL ETMIYORUZ — probe sonucu gelene
                // kadar bu 'load' tarayici hata sayfasi da olabilir.
            });
            iframe.addEventListener('error', function () { _iframeFail('error event'); });

            _iframeWatchTimer = setTimeout(function () {
                _iframeWatchTimer = null;
                if (_iframeLoaded) return;
                // Iframe halen yuklenmedi - kaynak unreachable kabul et, sonraki embed'e gec.
                _iframeFail('12sn icinde load yok');
            }, 12000);

            // Erisilebilirlik probe'u (8sn tavan). fetch/AbortController yoksa
            // eski davranisa dus: sadece no-load watchdog'u kalir.
            if (typeof window.fetch === 'function' && typeof window.AbortController === 'function') {
                var _probeCtl = new AbortController();
                var _probeTo  = setTimeout(function () { try { _probeCtl.abort(); } catch (e) {} }, 8000);
                fetch(s.src, {
                    method: 'GET',
                    mode: 'no-cors',
                    redirect: 'follow',
                    cache: 'no-store',
                    credentials: 'omit',
                    signal: _probeCtl.signal
                }).then(function () {
                    clearTimeout(_probeTo);
                    // Saglayiciya ulasildi (opak cevap). Iframe zaten yuklendiyse
                    // watchdog'a gerek yok.
                    if (_iframeLoaded && _iframeWatchTimer) {
                        clearTimeout(_iframeWatchTimer); _iframeWatchTimer = null;
                    }
                }).catch(function (err) {
                    clearTimeout(_probeTo);
                    // abort = 8sn'de cevap yok; reject = DNS/baglanti hatasi ya da
                    // kullanici tarafinda engel (blocker/DNS filtresi). Her iki
                    // durumda da iframe icerigi izlenebilir degil.
                    _iframeFail(err && err.name === 'AbortError' ? 'probe timeout' : 'probe reject');
                });
            }
        }
    }

    // Birlikte Izle / room mode entegrasyonu icin global expose
    window.czRenderServer = renderServer;

    /* ===== Tam ekranda kalarak sonraki bölüme geçiş =====
       Sorun: ArtPlayer tarayıcı fullscreen API'sini kendi $player (.art-video-player)
       elementine uygular. Bölüm değişiminde player yeniden kurulduğu için (renderServer
       eski $player'ı yok eder) fullscreen düşer; yeni sayfada kullanıcı-jesti olmadan
       tekrar fullscreen'e girilemez (tarayıcı engeli). Çözüm: geçiş anında fullscreen'i
       KALICI .player-wrap konteynerine taşı, player'ı onun İÇİNDE yeniden kur (FS düşmez),
       yeni player hazır olunca FS'i yeni $player'a geri ver (ArtPlayer durum/ikon
       tutarlılığı korunur). "Zaten fullscreen'deyken farklı elemana requestFullscreen"
       jest gerektirmediğinden (Chromium/Firefox) oto-geçişte de çalışır.
       KISIT: Yalnızca native (ArtPlayer) sonraki-bölümde uygulanır; saf iframe embed'lerin
       cross-origin fullscreen'ini kontrol edemeyiz → normal navigasyona güvenle düşer.
       iOS yalnızca <video> elementini fullscreen yaptığı için orada da fallback olur. */
    function czBrowserFsEl() {
        return document.fullscreenElement || document.webkitFullscreenElement
            || document.mozFullScreenElement || document.msFullscreenElement || null;
    }
    function czRequestFs(elm) {
        var fn = elm.requestFullscreen || elm.webkitRequestFullscreen
              || elm.mozRequestFullScreen || elm.msRequestFullscreen;
        if (!fn) return Promise.reject(new Error('fs-unsupported'));
        var r;
        try { r = fn.call(elm); } catch (e) { return Promise.reject(e); }
        return (r && typeof r.then === 'function') ? r : Promise.resolve();
    }

    // Fetch edilen bölüm sayfasının HTML'inden gömülü JS state'ini söker.
    // Sunucu zaten tüm imzalı URL'leri (np/hdfc/byse resolve token'ları) bu kullanıcının
    // oturumuna göre hazırladığı için yeniden imzalamaya gerek yok. Hata → null.
    function czParseEpisodePayload(html) {
        try {
            function b64Var(name) {
                // DİKKAT: PHP'nin dış json_encode'u base64'teki '/' karakterlerini '\/' diye
                // escape eder → tırnaklı literali önce JS string olarak çöz (JSON.parse),
                // sonra atob+JSON.parse (sayfanın kendisiyle birebir aynı sonuç).
                var m = html.match(new RegExp(name + '\\s*=\\s*JSON\\.parse\\(atob\\(("(?:\\\\.|[^"\\\\])*")\\)\\)'));
                return m ? JSON.parse(atob(JSON.parse(m[1]))) : null;
            }
            function strVar(name) {
                var m = html.match(new RegExp('var ' + name + '\\s*=\\s*("(?:\\\\.|[^"\\\\])*")\\s*;'));
                return m ? JSON.parse(m[1]) : null;
            }
            function intVar(name) {
                var m = html.match(new RegExp('var ' + name + '\\s*=\\s*(\\d+)\\s*;'));
                return m ? parseInt(m[1], 10) : null;
            }
            var serversP = b64Var('var servers');
            if (!Array.isArray(serversP) || !serversP.length) return null;
            var sblM = html.match(/var serversByLang\s*=\s*JSON\.parse\(atob\(("(?:\\.|[^"\\])*")\)\)/);
            var hlM  = html.match(/var hasLangBtns\s*=\s*(true|false)/);
            var dlM  = html.match(/var defaultLang\s*=\s*("(?:\\.|[^"\\])*"|null)/);
            var tM   = html.match(/<title[^>]*>([\s\S]*?)<\/title>/i);
            return {
                servers:       serversP,
                serversByLang: sblM ? JSON.parse(atob(JSON.parse(sblM[1]))) : null,
                hasLangBtns:   hlM ? (hlM[1] === 'true') : null,
                defaultLang:   dlM ? JSON.parse(dlM[1]) : null,
                epId:          intVar('epId'),
                epLabel:       strVar('epLabel'),
                epNoForSkip:   intVar('epNoForSkip'),
                nextUrl:       strVar('nextUrl'),
                title:         tM ? tM[1].replace(/&amp;/g, '&').replace(/&#0?39;/g, "'").replace(/&quot;/g, '"').trim() : null
            };
        } catch (e) {
            return null;
        }
    }

    var _czSwitchInFlight = false;
    // Fetch+parse+yerinde-rebuild. Başarılı → true. Çağıran false'ta normal navigasyona düşer.
    function czSwitchEpisodeInPlace(url) {
        return new Promise(function (resolve) {
            if (!url || _czSwitchInFlight || !czBrowserFsEl()) { resolve(false); return; }
            _czSwitchInFlight = true;
            var done = function (v) { _czSwitchInFlight = false; resolve(v); };
            fetch(url, { credentials: 'same-origin' })
                .then(function (r) { return r.ok ? r.text() : null; })
                .then(function (html) {
                    if (!html) throw new Error('fetch');
                    var p = czParseEpisodePayload(html);
                    if (!p) throw new Error('parse');
                    // Sadece native sonraki-bölüm: iframe embed'in FS'ini kontrol edemeyiz.
                    if (NATIVE_TYPES.indexOf(p.servers[0].type) === -1) throw new Error('iframe-next');
                    var wrap = document.querySelector('.player-wrap');
                    if (!wrap) throw new Error('no-wrap');
                    // FS'i kalıcı wrap'e taşı — rebuild $player'ı yok ederken FS düşmesin.
                    return czRequestFs(wrap).then(function () { return p; });
                })
                .then(function (p) {
                    if (!p) { done(false); return; }
                    // Closure state'i yeni bölüme çevir — renderServer bunları okur.
                    servers          = p.servers;
                    serversByLang    = p.serversByLang || serversByLang;
                    if (p.hasLangBtns !== null) hasLangBtns = p.hasLangBtns;
                    if (p.defaultLang)          defaultLang = p.defaultLang;
                    // 2026-08-03: kullanicinin SECTIGI dil korunur ve servers o dile
                    // gore FILTRELENIR. Eskiden burada kosulsuz `currentLang =
                    // defaultLang` vardi, ustelik `servers` p.servers (dile gore
                    // filtrelenmemis GLOBAL priority listesi) olarak kaliyordu ->
                    // renderServer(0) listenin tepesindekini aciyordu. Naruto'da tau
                    // (altyazi) ile CVD (dublaj) ayni priority'de (100) oldugu icin
                    // FS'te "sonraki bolum"e basan dublaj izleyicisi altyaziya
                    // dusuyordu; FS'ten cikista sayfa yeniden yuklenip sunucu tarafi
                    // filtre calisinca dublaja "geri donuyordu". activate() ve
                    // switchLang() ile ayni filtre artik burada da uygulanir.
                    var wantLang = currentLang;
                    if (!wantLang || wantLang === 'any'
                        || !(serversByLang && serversByLang[wantLang]
                             && serversByLang[wantLang].length)) {
                        wantLang = defaultLang;      // yeni bolumde o dil yoksa geri dus
                    }
                    currentLang = wantLang;
                    if (wantLang && wantLang !== 'any'
                        && serversByLang && serversByLang[wantLang]
                        && serversByLang[wantLang].length) {
                        servers = serversByLang[wantLang];
                    }
                    // Dil butonlari FS'te gorunmese de state'i senkron tut (FS'ten
                    // cikista sayfa yeniden yuklenene kadar dogru buton aktif kalsin).
                    try {
                        var _dBtn = document.getElementById('langDubBtn');
                        var _sBtn = document.getElementById('langSubBtn');
                        if (_dBtn) _dBtn.classList.toggle('is-active', currentLang === 'dub');
                        if (_sBtn) _sBtn.classList.toggle('is-active', currentLang === 'sub');
                    } catch (e) {}
                    current          = 0;
                    if (p.epId)             epId        = p.epId;
                    if (p.epLabel !== null) epLabel     = p.epLabel;
                    if (p.epNoForSkip)      epNoForSkip = p.epNoForSkip;
                    nextUrl          = (typeof p.nextUrl === 'string') ? p.nextUrl : '';
                    _serverAttempted = {};                 // yeni bölüm için fallback izini sıfırla
                    window.czPendingFsRestore = true;       // yeni player hazır olunca FS'i geri al
                    window.czEpSwitchedInFs   = true;       // FS'ten çıkışta sayfayı senkronla
                    window.czCurrentEpUrl     = url;
                    try { history.pushState({}, '', url); } catch (e) {}
                    if (p.title) { try { document.title = p.title; } catch (e) {} }
                    renderServer(0);                        // wrap FS olduğu için FS korunur
                    if (typeof window.czShowToast === 'function') {
                        window.czShowToast('Sonraki bölüme geçildi.', 'success');
                    }
                    done(true);
                })
                .catch(function () { done(false); });
        });
    }

    // Sonraki bölüme git: FS'deysek yerinde geç (FS koru), olmazsa/başarısızsa normal navigasyon.
    // Birlikte İzle (room) modunda: yerinde geçiş watch-party.js'i yeni bölüme göre
    // yeniden başlatmadığı için oda senkronu bozulur → eski davranış (tam navigasyon).
    function czGoToNextEpisode(url) {
        url = url || nextUrl;
        if (!url) return;
        var inRoom = (typeof window.czRoomActive === 'function') && window.czRoomActive();
        if (czBrowserFsEl() && !inRoom) {
            czSwitchEpisodeInPlace(url).then(function (ok) {
                if (!ok) window.location.href = url;   // native değil/parse hatası vb. → tam yükle
            });
        } else {
            window.location.href = url;
        }
    }
    window.czGoToNextEpisode = czGoToNextEpisode;

    // "Sıradaki Bölüm" overlay butonu (.cz-next-ep): her renderServer'da yeniden
    // oluşturulduğu için kalıcı .player-wrap'e delege dinleyici bağla (bir kez).
    (function () {
        var pw = document.querySelector('.player-wrap');
        if (!pw) return;
        pw.addEventListener('click', function (e) {
            var a = (e.target && e.target.closest) ? e.target.closest('.cz-next-ep') : null;
            if (!a) return;
            // Artik <a> degil <button> (bkz. layer html'indeki reklam katmani notu),
            // yani tarayici navigasyonu YOK -> tam ekran disinda da biz goturuyoruz.
            e.preventDefault();
            e.stopImmediatePropagation();
            czGoToNextEpisode(a.getAttribute('data-next-url') || nextUrl);
        }, true);
    })();

    // FS'ten ÇIKINCA (ESC/çıkış butonu): in-place geçiş yapıldıysa sayfayı geçerli
    // bölüme senkronla (yorumlar/başlık/komşu butonlar otursun). Geçiş sırasındaki ara
    // fullscreenchange'leri (FS elemanı wrap↔$player değişimi) gerçek çıkışla karıştırmamak
    // için kısa debounce ile son durumu kontrol et.
    (function () {
        var t = null;
        function onFsChange() {
            if (t) clearTimeout(t);
            t = setTimeout(function () {
                t = null;
                if (czBrowserFsEl()) return;            // hâlâ FS → çıkış değil
                if (!window.czEpSwitchedInFs) return;   // in-place geçiş olmadıysa dokunma
                window.czEpSwitchedInFs = false;
                // Yeniden yüklenince poster kapısı otomatik açılsın (ekstra tıklama olmasın).
                try { sessionStorage.setItem('cz-gate-autostart', '1'); } catch (e) {}
                window.location.href = window.czCurrentEpUrl || location.href;
            }, 250);
        }
        document.addEventListener('fullscreenchange', onFsChange);
        document.addEventListener('webkitfullscreenchange', onFsChange);
        document.addEventListener('mozfullscreenchange', onFsChange);
    })();

    // Player değişiminden önce mevcut native player'ın anlık zamanını
    // cz-resume-{epId}'ye yaz - yeni player metadata yüklenince czRestore
    // bu key'i okuyup otomatik seek yapar. İframe time'ı bilinmediği için
    // sadece native (ArtPlayer) kaynaklarda anlamlı; iframe → native geçişte
    // throttled save'in son yazdığı değer kullanılır.
    // Bitime <60sn kalan konumu KAYDETME: czSaveResume ile ayni kural. Aksi
    // halde dublaj->altyazi gecisinde bolum sonuna yakin konum yazilip yeni
    // player onu geri yukluyor, video hemen bitiyor ve "Oto Sonraki Bolum"
    // acikken kullanici sonraki bolume firlatiliyordu.
    function persistCurrentTimeForResume() {
        if (!epId) return;
        try {
            var p = window.czPlayer;
            var t = (p && p.currentTime) || 0;
            if (!(t > 1 && isFinite(t))) return;
            var d = p && p.duration;
            if (d && isFinite(d) && t > d - 60) {
                localStorage.removeItem('cz-resume-' + epId);
                return;
            }
            localStorage.setItem('cz-resume-' + epId, String(t));
        } catch (e) {}
    }

    // ===== Poster gate: ilk tikta player'i lazy mount eder =====
    // Initial HTML'de iframe veya ArtPlayer yok - sadece poster + play butonu.
    // Tiklayinca renderServer(0) ilk embed'i mount eder.
    (function () {
        var gate = document.getElementById('czPosterGate');
        if (!gate || !servers.length) return;
        var activated = false;
        // czInitPlayer SONRAKİ <script> bloğunda tanımlı. Kullanıcı poster hâlâ
        // dönerken (sayfa parse edilirken) play tuşuna erken tıklarsa fonksiyon
        // henüz yok olabilir → renderServer native embed'de "Oynatıcı yüklenemedi"
        // fallback'ine düşerdi. Hazır olana kadar bekle (poster spinner dönmeye devam eder).
        function whenPlayerReady(cb) {
            if (typeof czInitPlayer === 'function') { cb(); return; }
            if (document.readyState === 'loading') {
                document.addEventListener('DOMContentLoaded', function () { whenPlayerReady(cb); }, { once: true });
                return;
            }
            // readyState interactive/complete ama fonksiyon hâlâ yok (çok nadir) → kısa poll.
            var tries = 0;
            var t = setInterval(function () {
                if (typeof czInitPlayer === 'function' || ++tries > 100) {
                    clearInterval(t);
                    cb();
                }
            }, 30);
        }
        function activate() {
            if (activated) return;
            activated = true;
            gate.classList.add('is-loading');
            // defaultLang filtresi: dub/sub varsa servers'i lang listesiyle baslat,
            // yoksa global sirayi koru (any).
            if (defaultLang !== 'any' && serversByLang[defaultLang] && serversByLang[defaultLang].length) {
                servers = serversByLang[defaultLang];
                current = 0;
            }
            whenPlayerReady(function () {
                renderServer(0);
                if (gate.parentNode) gate.parentNode.removeChild(gate);
            });
        }
        gate.addEventListener('click', activate);
        gate.addEventListener('keydown', function (e) {
            if (e.key === 'Enter' || e.key === ' ') {
                e.preventDefault();
                activate();
            }
        });
        // Tam ekranda bölüm değiştirip FS'ten çıkınca sayfa yeniden yüklenir; az önce
        // izlemekte olan kullanıcı için poster kapısını OTOMATİK aç (player'ı mount et,
        // ekstra tıklama gerekmesin). Yalnızca kullanıcının kendi koyduğu sessionStorage
        // işaretiyle tetiklenir → bot'lar bu işareti taşımaz, normal kapıyı görür (cloaking
        // değil; herkese aynı HTML, sadece kullanıcı-eylemine bağlı JS davranışı).
        try {
            if (sessionStorage.getItem('cz-gate-autostart') === '1') {
                sessionStorage.removeItem('cz-gate-autostart');
                // activate() artık czInitPlayer hazırlığını whenPlayerReady ile kendisi
                // beklediği için doğrudan çağrılabilir (eski DOMContentLoaded sarmalı gereksiz).
                activate();
            }
        } catch (e) {}
    })();

    if (altBtn && servers.length >= 2) {

        altBtn.addEventListener('click', function () {
            // Sadece ilk SERVER_TOGGLE_MAX (=2) embed arasinda dongusel: 0 <-> 1
            var cycleSize = Math.min(servers.length, SERVER_TOGGLE_MAX);
            // Eger current cycleSize disindaysa (3+), 0'a sifirla
            if (current >= cycleSize) current = 0;
            var nextIdx = (current + 1) % cycleSize;

            // Birlikte Izle aktifse: misafir degistiremez, host degistirince
            // server'a bildir; server diger uyelere player_changed broadcast eder.
            var inRoom = typeof window.czRoomActive === 'function' && window.czRoomActive();
            if (inRoom) {
                var isHost = typeof window.czRoomIsHost === 'function' && window.czRoomIsHost();
                if (!isHost) {
                    toast('Sadece host oynatıcıyı değiştirebilir.', 'warn');
                    return;
                }
                // Room mode'da non-native embed'e geciste sync bozulur (iframe'in
                // currentTime'i bilinmiyor + cz-player-ready firlenmiyor); host
                // istese de bu gecisi engelliyoruz.
                var targetSv = servers[nextIdx];
                if (!targetSv || NATIVE_TYPES.indexOf(targetSv.type) === -1) {
                    toast('Bu oynatıcı Birlikte İzle ile uyumsuz, geçilemiyor.', 'warn');
                    return;
                }
                // Host'un mevcut zamanini al - yeni embed ayni saniyeden devam etsin
                var curTime = 0;
                try { curTime = (window.czPlayer && window.czPlayer.currentTime) || 0; } catch (e) {}
                persistCurrentTimeForResume();
                if (typeof window.czRoomChangePlayer === 'function') {
                    window.czRoomChangePlayer(nextIdx, curTime);
                }
                renderServer(nextIdx, { startTime: curTime });
            } else {
                persistCurrentTimeForResume();
                renderServer(nextIdx);
            }
            toast(nextIdx === 0 ? 'Ana oynatıcıya geçildi.' : 'Alternatif oynatıcıya geçildi.', 'success');
        });

        /* ===== Altyazı / Dublaj dil butonlari =====
           webteizle film embed'lerinde lang='dub'/'sub' set edilir; bu butonlar
           player'i ilgili dile siler. Server listesi swap edilerek alt-player
           da yeni dil icinde calisir. Kullanici tercihi localStorage'da. */
        if (hasLangBtns) {
            var dubBtn = document.getElementById('langDubBtn');
            var subBtn = document.getElementById('langSubBtn');
            var LANG_PREF_KEY = 'cz-lang-pref';

            function setLangActiveState(lang) {
                if (dubBtn) dubBtn.classList.toggle('is-active', lang === 'dub');
                if (subBtn) subBtn.classList.toggle('is-active', lang === 'sub');
            }

            function switchLang(newLang, opts) {
                if (newLang !== 'dub' && newLang !== 'sub' && newLang !== 'any') return;
                var newList = serversByLang[newLang] || [];
                if (!newList.length) return;
                if (newLang === currentLang && !(opts && opts.force)) return;

                // Eger su anki oynayan embed yeni listenin de 1. siradaysa
                // (rapidvid gibi 'any' lang dual-source'lar her iki listede de
                // top-priority) gereksiz player reload'u atla; sadece state +
                // buton aktif rengi + localStorage guncelle. Kullanici lang
                // butonuna basinca rapidvid icindeki kendi quick-mode
                // (Türkçe Dublaj / Türkçe Altyazılı) ile dil degisimi yapsin.
                var curSv = servers[current];
                var curEmbedId = curSv && curSv.embedId;
                var sameTop = curEmbedId && newList[0] && newList[0].embedId === curEmbedId;

                currentLang = newLang;
                servers = newList;
                current = 0;
                setLangActiveState(newLang);
                try { localStorage.setItem(LANG_PREF_KEY, newLang); } catch (e) {}

                if (sameTop) {
                    toast(newLang === 'dub' ? 'Dublaj seçildi.'
                                             : (newLang === 'sub' ? 'Altyazı seçildi.' : ''),
                          'success');
                    return;
                }

                var startTime = 0;
                try {
                    startTime = (window.czPlayer && window.czPlayer.currentTime) || 0;
                } catch (e) {}
                persistCurrentTimeForResume();

    function bindToggle(btn, opts) {
        if (!btn) return;
        var state = !!opts.initial;
        setIcon(btn, state);

        btn.addEventListener('click', function () {
            if (!isLoggedIn) { showLoginRequired(); return; }

            var willBeOn = !state;
            // Optimistik UI
            state = willBeOn;
            setIcon(btn, state);
            if (typeof opts.onToggle === 'function') opts.onToggle(willBeOn);
            toast(willBeOn ? opts.onMsg : opts.offMsg,
                  willBeOn ? opts.onKind : opts.offKind);

            fetch(opts.url, {
                method: 'POST',
                credentials: 'same-origin',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(opts.payload)
            })
            .then(function (res) {
                if (res.status === 401) {
                    state = !willBeOn;
                    setIcon(btn, state);
                    if (typeof opts.onToggle === 'function') opts.onToggle(state);
                    showLoginRequired();
                }
                return res.json().catch(function(){ return null; });
            })
            .catch(function () {
                // Hata durumunda eski state'e dön
                state = !willBeOn;
                setIcon(btn, state);
                if (typeof opts.onToggle === 'function') opts.onToggle(state);
            });
        });
    }

         duration:    bolum suresi (eslik kontrolu icin; %75 esigi)

    if (epId > 0) {
        var hbUrl = '/api/simdi-izleniyor/';
        var clearBody = JSON.stringify({ clear: true });
        // app.js fetch wrapper'ı defer yüklendiği için inline heartbeat'in ilk
        // çağrısında henüz aktif olmayabilir. CSRF token'ı meta'dan kendimiz okuyup
        // tüm fetch/sendBeacon çağrılarına manuel ekliyoruz (race condition fix).
        var __hbCsrf = (function () {
            var m = document.querySelector('meta[name="csrf-token"]');
            return m ? m.getAttribute('content') || '' : '';
        })();
        function sendHeartbeat() {
            if (document.hidden) return;
            var ct = 0, du = 0;
            try {
                var p = window.czArtPlayer;
                if (p) {
                    ct = Math.max(0, Math.floor(p.currentTime || 0));
                    du = Math.max(0, Math.floor(p.duration   || 0));
                }
            } catch (e) {}
            var activeEmbedId = 0;
            try {
                var idx = (typeof current === 'number') ? current : 0;
                if (servers && servers[idx] && servers[idx].embedId) {
                    activeEmbedId = Number(servers[idx].embedId) || 0;
                }
            } catch (e) {}
            var body = JSON.stringify({
                episode_id:   epId,
                current_time: ct,
                duration:     du,
                visible:      document.hidden ? 0 : 1,
                embed_id:     activeEmbedId
            });
            fetch(hbUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json', 'X-CSRF-Token': __hbCsrf },
                body: body,
                keepalive: true
            }).catch(function () {});
        }
        function sendClear() {
            // sendBeacon X-CSRF-Token taşıyamaz (header ekleyemez); CSRF'i body'ye
            // gömerek gönderiyoruz - Auth::requireCsrf body._csrf'ı da kontrol ediyor.
            try {
                var beaconBody = JSON.stringify({ clear: true, _csrf: __hbCsrf });
                var blob = new Blob([beaconBody], { type: 'application/json' });
                if (navigator.sendBeacon && navigator.sendBeacon(hbUrl, blob)) return;
            } catch (e) {}
            fetch(hbUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json', 'X-CSRF-Token': __hbCsrf },
                body: clearBody,
                keepalive: true
            }).catch(function () {});
        }
        sendHeartbeat();
        setInterval(sendHeartbeat, 30000);
        // Sekme arka plana geçince heartbeat'i SİLMİYORUZ - başka sekmeye baksa
        // bile bölüm sayfası açık olduğu sürece "izliyor" sayılması doğru.
        // 90sn'lik pencere kayıt eskidiğinde otomatik gizler.
        // Sayfaya geri dönüldüğünde anında bir heartbeat atalım ki sayım canlı kalsın.
        document.addEventListener('visibilitychange', function () {
            if (!document.hidden) sendHeartbeat();
        });
        // Sadece sayfa gerçekten kapanırken kaydı temizle.
        window.addEventListener('pagehide', sendClear);
        window.addEventListener('beforeunload', sendClear);
    }

    function detectBrowser() {

        if (/Edg\//.test(ua))           return 'Edge';
        if (/OPR\//.test(ua))           return 'Opera';
        if (/Chrome\//.test(ua) && !/Chromium/.test(ua)) return 'Chrome';

        return 'Diğer';

    function detectOS() {
        var ua = navigator.userAgent;
        if (/Windows NT 10\.0/.test(ua)) return 'Windows 10/11';
        if (/Windows NT 6\.3/.test(ua))  return 'Windows 8.1';
        if (/Windows NT 6\.2/.test(ua))  return 'Windows 8';
        if (/Windows NT 6\.1/.test(ua))  return 'Windows 7';
        if (/Windows/.test(ua))          return 'Windows';
        if (/Android/.test(ua))          return 'Android';
        if (/iPhone|iPad|iPod/.test(ua)) return 'iOS';
        if (/Mac OS X/.test(ua))         return 'macOS';
        if (/Linux/.test(ua))            return 'Linux';

    /* Bildiri gönderilirken otomatik teknik teşhis snapshot'ı üretir.
       Kullanıcı açıklama yazmasa bile admin "ne çıkmış" görebilsin diye:
       aktif sunucu, native <video> hata kodu, hls/console hataları, ağ durumu.
       Her erişim ayrı try/catch'te — biri patlasa diğerleri yine toplanır. */
    function buildDiagnostics() {
        var d = {};
        try { d.url = location.href; } catch (e) {}
        try { d.at  = new Date().toISOString(); } catch (e) {}
        try { d.online = navigator.onLine; } catch (e) {}
        try {
            var nc = navigator.connection || navigator.mozConnection || navigator.webkitConnection;
            if (nc) d.conn = { type: nc.effectiveType, down: nc.downlink, rtt: nc.rtt, save: nc.saveData };
        } catch (e) {}
        try {
            d.screen = window.innerWidth + 'x' + window.innerHeight +
                ' @' + (window.devicePixelRatio || 1) + 'x (ekran ' + screen.width + 'x' + screen.height + ')';
        } catch (e) {}

        // Aktif sunucu + mevcut sunucu listesi (servers/current outer IIFE'de).
        try {
            if (typeof servers !== 'undefined' && servers) {
                var idx = (typeof current === 'number') ? current : 0;
                var s = servers[idx] || {};
                d.activeServer = {
                    idx: idx + 1, total: servers.length,
                    type: s.type || null, label: s.label || null,
                    lang: s.lang || null, embedId: s.embedId || null
                };
                d.servers = servers.map(function (x) {
                    return (x.type || '?') + ':' + (x.label || '?') + (x.lang ? '(' + x.lang + ')' : '');
                });
                // Saf iframe player → konsolu cross-origin, okunamaz; admin'e not düş.
                if (s.type === 'iframe') d.note = 'Player iframe (cross-origin) — iç konsol okunamaz.';
            }
        } catch (e) {}

        // Native ArtPlayer / <video> hata durumu.
        try {
            var art = window.czArtPlayer;
            if (art) {
                var v = art.video || (art.template && art.template.$video) || null;
                var vd = { player: 'artplayer' };
                if (v) {
                    var errMap = { 1: 'ABORTED', 2: 'NETWORK', 3: 'DECODE', 4: 'SRC_NOT_SUPPORTED' };
                    if (v.error) vd.error = { code: v.error.code, name: errMap[v.error.code] || '?', message: v.error.message || '' };
                    vd.networkState = v.networkState;   // 3 = NETWORK_NO_SOURCE
                    vd.readyState   = v.readyState;     // 0 = HAVE_NOTHING
                    vd.currentTime  = Math.round((v.currentTime || 0) * 10) / 10;
                    vd.duration     = isFinite(v.duration) ? Math.round(v.duration) : null;
                    vd.paused       = v.paused; vd.ended = v.ended;
                    try { vd.buffered = v.buffered && v.buffered.length ? Math.round(v.buffered.end(v.buffered.length - 1)) : 0; } catch (e2) {}
                    try { vd.currentSrc = (v.currentSrc || '').slice(0, 200); } catch (e2) {}
                }
                vd.hls = !!art.hls;
                d.video = vd;
            } else {
                d.video = { player: 'iframe-or-none' };
            }
        } catch (e) {}

        // Yakalanan konsol/hata logları (ring buffer).
        try {
            if (window.__czDiag && window.__czDiag.buf) {
                d.log = window.__czDiag.buf.map(function (x) {
                    return '+' + (x.ms / 1000).toFixed(1) + 's [' + x.l + '] ' + x.m;
                });
            }
        } catch (e) {}

        try { return JSON.stringify(d); }
        catch (e) { return JSON.stringify({ err: 'diag_serialize_failed', url: (location && location.href) || '' }); }
    }

var CZ_BW_CACHE_TTL_MS = 30 * 60 * 1000;
function czGetCachedBw() {
    try {
        var c = JSON.parse(localStorage.getItem('cz-bw') || 'null');
        if (c && typeof c.mbps === 'number' && (Date.now() - c.t) < CZ_BW_CACHE_TTL_MS) {

// HTTP probe: 2 MB Range fetch, ilk byte sonrasi 1.5 saniyelik sample.
// TTFB hariç tutulur - sadece veri akış hızı ölçülür → güvenilir Mbps.
function czProbeBandwidth(url) {
    return new Promise(function (resolve) {
        if (!window.fetch || typeof AbortController === 'undefined') return resolve(null);
        var ctrl = new AbortController();
        var safety = setTimeout(function () { try { ctrl.abort(); } catch (e) {} resolve(null); }, 7000);

        fetch(url, {
            headers: { 'Range': 'bytes=0-2097151' },
            cache: 'no-store',
            credentials: 'same-origin',
            signal: ctrl.signal
        })
        .then(function (resp) {
            if (resp.status !== 206 && !resp.ok) throw new Error('probe http ' + resp.status);
            if (!resp.body || !resp.body.getReader) throw new Error('no streaming body');
            var reader = resp.body.getReader();
            var bytes = 0;
            var firstByteAt = null;
            var done = false;

            function read() {
                return reader.read().then(function (chunk) {
                    if (done) return;
                    if (chunk.done) {
                        // Stream bitti; var olan ölçümle hesapla
                        if (firstByteAt !== null) {
                            var el = (performance.now() - firstByteAt) / 1000;
                            if (el > 0.05 && bytes > 0) {
                                done = true;
                                clearTimeout(safety);
                                resolve(((bytes * 8) / el) / 1e6);
                                return;
                            }
                        }
                        clearTimeout(safety);
                        resolve(null);
                        return;
                    }
                    if (firstByteAt === null) firstByteAt = performance.now();
                    bytes += chunk.value.length;
                    var elapsed = performance.now() - firstByteAt;
                    if (elapsed >= 1500) {
                        done = true;
                        try { ctrl.abort(); } catch (e) {}
                        clearTimeout(safety);
                        var elapsedSec = elapsed / 1000;
                        resolve(((bytes * 8) / elapsedSec) / 1e6);
                        return;
                    }
                    return read();
                }).catch(function () {
                    if (done) return;
                    done = true;
                    clearTimeout(safety);
                    if (firstByteAt !== null && bytes > 65536) {
                        var elapsedSec = (performance.now() - firstByteAt) / 1000;
                        resolve(((bytes * 8) / elapsedSec) / 1e6);
                    } else {
                        resolve(null);
                    }
                });
            }
            return read();
        })
        .catch(function () {
            clearTimeout(safety);
            resolve(null);
        });
    });
}

// Bandwidth tahmin: cache → probe → navigator.connection → cömert default.
// probeUrl = manifest'ten en küçük kalite URL'i (server yükü min).
function czEstimateBandwidth(probeUrl) {
    var cached = czGetCachedBw();
    if (cached !== null) {
        return Promise.resolve({ mbps: cached, source: 'cache' });
    }
    return czProbeBandwidth(probeUrl).then(function (probed) {
        if (probed !== null && probed > 0) {
            czSaveBw(probed);
            return { mbps: probed, source: 'probe' };
        }
        // Fallback: navigator.connection
        try {
            var nc = navigator.connection || navigator.mozConnection || navigator.webkitConnection;
            if (nc && typeof nc.downlink === 'number' && nc.downlink > 0) {
                czSaveBw(nc.downlink);
                return { mbps: nc.downlink, source: 'connection-api' };
            }
        } catch (e) {}
        // Bilinmiyor - cömert default (720P'ye yetecek + biraz yukari, 1080P'yi de elemiyor).
        return { mbps: 3.0, source: 'default' };
    });
}

// Mbps → q. qualities = manifest entry'leri (en yüksekten en düşüğe sıralı).
// '<sayi>p' label'inin piksel yuksekligi; format tutmuyorsa 0 ('high', 'fullhd'
// gibi Dzen/ok.ru enum'lari icin 0 doner - onlar CZ_QUALITY_MIN_MBPS'te zaten var).
function czHeightFromLabel(label) {
    var m = String(label || '').toLowerCase().match(/^(\d{2,4})p$/);
    return m ? parseInt(m[1], 10) : 0;
}

// CZ_QUALITY_MIN_MBPS tablosunda olmayan ama '<sayi>p' olan label'lar icin
// (tau re-encode: 476p, 416p, 384p...) esigi yukseklikten turet.
function czMinMbpsFromHeight(label) {
    var h = czHeightFromLabel(label);
    if (h <= 0)   return null;
    if (h >= 1080) return 3.5;
    if (h >= 720)  return 1.8;
    if (h >= 480)  return 1.0;
    if (h >= 360)  return 0.6;
    return 0.3;
}

function czPickQuality(qualities, mbps) {
    if (!qualities || !qualities.length) return null;
    for (var i = 0; i < qualities.length; i++) {
        var q = qualities[i].q;
        var min = CZ_QUALITY_MIN_MBPS[q];
        // Tablo disi sayisal label'lar ABR'de hic secilemeyip her zaman en dusuge
        // dusuyordu; esigi yukseklikten turetince normal varyant gibi davraniyor.
        if (typeof min !== 'number') min = czMinMbpsFromHeight(q);
        if (typeof min !== 'number') continue;
        if (mbps >= min) return q; // sıralı listede ilk uyan = en yüksek uyan
    }
    return qualities[qualities.length - 1].q;
}

// Upstream API cagrisini SANDBOX IFRAME icinden yapar - Origin: null gonderilir.
// Boylece upstream loglarinda "https://cizgimax.online" Origin'i GORUNMEZ;
// onun yerine sandbox iframe'lerin yaygin "null" Origin'i gorunur (Stack Overflow,
// CodePen, JSFiddle vs. ile aynilastirilabilir, fingerprint olusmaz).
//
// Akis:
//   1) <iframe sandbox="allow-scripts" srcdoc="..."> yarat (display:none)
//      sandbox flag'inde "allow-same-origin" YOK → opaque origin → Origin: null
//   2) postMessage ile iframe'e embed_id gonder
//   3) Iframe icindeki JS upstream API'yi cagirir (Origin: null), sonucu postMessage ile yollar
//   4) Parent sonucu alir, iframe'i temizler
function czNpApiViaSandbox(embedId) {
    return new Promise(function (resolve, reject) {
        var srcdoc = window.CZ_NP_SANDBOX_SRCDOC || '';
        if (!srcdoc) return reject(new Error('sandbox srcdoc missing'));

        var requestId = 'r' + Date.now() + '_' + Math.random().toString(36).slice(2, 8);

        var iframe = document.createElement('iframe');
        iframe.setAttribute('sandbox', 'allow-scripts');  // allow-same-origin YOK → opaque origin
        iframe.style.cssText = 'display:none;position:absolute;width:0;height:0;border:0';
        iframe.srcdoc = srcdoc;

        var settled = false;
        var timeout = setTimeout(function () {
            if (settled) return;
            settled = true;
            cleanup();
            reject(new Error('np api timeout'));
        }, 15000);

        function cleanup() {
            clearTimeout(timeout);
            window.removeEventListener('message', onMessage);
            if (iframe.parentNode) iframe.parentNode.removeChild(iframe);
        }
        function onMessage(e) {
            var d = e.data;
            if (!d || d.type !== 'np-fetch-result' || d.requestId !== requestId) return;
            if (settled) return;
            settled = true;
            cleanup();
            if (d.error) reject(new Error(d.error));
            else resolve(d.data);
        }
        window.addEventListener('message', onMessage);

        iframe.addEventListener('load', function () {
            try {
                iframe.contentWindow.postMessage(
                    { type: 'np-fetch', id: embedId, requestId: requestId },
                    '*'  // sandbox iframe origin null - targetOrigin '*' zorunlu
                );
            } catch (err) {
                if (settled) return;
                settled = true;
                cleanup();
                reject(err);
            }
        });

        document.body.appendChild(iframe);
    });
}

// Native opaque resolve flow (client-side).
//   1) PHP /api/np-resolve/?t=<token> → {"id": "<embed_id>"}
//      (bizim sunucu upstream'e gitmiyor - sadece DB lookup)
//   2) Sandbox iframe (Origin: null) → upstream API → urls[]
//      (kullanici browser'i, Origin: null, Referer yok → "cizgimax" hicbir yerde gecmez)
//   3) POST /api/sign-np/ → upstream URL'leri HMAC ile imzalanir (server URL'i
//      fetch ETMEZ, sadece imzalar). ArtPlayer src'i /api/stream/dzen/?s=<sig>
//      olur; info panel'de upstream URL gozukmez. Sign fail olursa direct URL
//      fallback (oynar ama info panel'de upstream gozukur - graceful degrade).
//   4) MP4'ler 302 redirect ile CDN'den dogrudan oynar - Sibnet pattern,
//      bandwidth bizden gecmez. Referer same-origin meta'si CDN'e Referer
//      gondermez → upstream log'larinda "cizgimax" hicbir header'da yok.
function czResolveNpPlayer(el, resolveUrl, title, epLabel, nextUrl, malId, epNo, epId) {
    // ⚠️ Sabit kalite beyaz listesi (eski NP_QUALITY_ORDER = ['1080p','720p','480p'])
    // KULLANILMIYOR: tau re-encode'lari '476p', '416p', '384p', '360p' gibi
    // standart-disi yukseklikler donuyor, whitelist bunlari eleyince qualities[]
    // bos kaliyor ve "Desteklenen kalite bulunamadı" → tek kaynakli bolum tamamen
    // olu oluyordu (One Piece S1B437 = tek kaynak '476p'). Artik '<sayi>p'
    // formatindaki her label kabul edilip sayisal olarak buyukten kucuge siralanir.
    var NP_DEFAULT = '720p';

    function showError(msg) {
        // Native (tau/np) resolve başarısız → statik mesajla kullanıcıyı kilitlemek
        // yerine sonraki denenmemiş sunucuya OTOMATİK geç (örn. dzen.ru/sibnet alternatifi).
        // Tau kaynağı CANLI olsa bile animecix rate-limit (429) ile resolve ara sıra
        // patlar; fallback olmadan kullanıcı çalışan alternatif dururken kilitleniyordu
        // (Bleach S1E1 hata bildirimi). Alternatif kalmadıysa czHandleServerFailure kendi
        // "Video yüklenemedi" + Yenile overlay'ini basar.
        if (typeof window.czHandleServerFailure === 'function') {
            if (window.console) console.warn('[np] resolve başarısız → sonraki sunucuya geçiliyor: ' + msg);
            window.czHandleServerFailure();
            return;
        }
        el.innerHTML = '<div style="background:#000;color:#888;padding:60px;text-align:center"><p>' +
                       msg.replace(/[<>&]/g, function (c) { return {'<':'&lt;','>':'&gt;','&':'&amp;'}[c]; }) +
                       '</p></div>';
    }

    // Embed token'i resolveUrl'in ?t= parametresinden cek - sign endpoint'i ayni
    // token'i kullaniyor (token zaten HMAC-signed embed_id'yi tasiyor).
    var tokenMatch = String(resolveUrl).match(/[?&]t=([^&]+)/);
    var embedToken = tokenMatch ? decodeURIComponent(tokenMatch[1]) : '';

    // Sign step - urls[].url -> /api/stream/dzen/?s=<sig> donusturur, captions[] de
    // ayni POST'ta /api/np-sub/?s=<sig> formatinda imzalanip donerse ArtPlayer
    // subtitle URL'i olarak kullanilir. Fail durumunda original URL'leri kullanir
    // (player calismaya devam eder, sadece subtitle yoksa CC menusu gozukmez).
    function maybeSignQualities(qualities, captions) {
        if (!embedToken || !qualities.length) return Promise.resolve({ qualities: qualities, subs: [] });
        var payload = {
            token: embedToken,
            urls: qualities.map(function (q) { return { q: q.q, url: q.url }; })
        };
        if (Array.isArray(captions) && captions.length) {
            payload.captions = captions.map(function (c) {
                return { lang: c.lang, name: c.name, url: c.url };
            });
        }
        // CSRF: Router /api/* POST'lar icin global X-CSRF-Token gerektirir
        // (Auth::requireCsrf). MP4-only flow csrf fail'inde graceful degrade
        // ediyordu (direct URL'e dusuyor), ama subtitle proxy server-side fetch
        // gerektirdigi icin sign zorunlu — token yoksa subList olusturulamaz.
        var __csrfMeta = document.querySelector('meta[name="csrf-token"]');
        var __csrfTok  = __csrfMeta ? __csrfMeta.getAttribute('content') || '' : '';
        return fetch('/api/sign-np/', {
            method: 'POST',
            credentials: 'same-origin',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRF-Token': __csrfTok
            },
            body: JSON.stringify(payload)
        })
        .then(function (r) { return r.ok ? r.json() : null; })
        .catch(function () { return null; })
        .then(function (signData) {
            var subs = (signData && Array.isArray(signData.signedSubs)) ? signData.signedSubs : [];
            if (!signData || !Array.isArray(signData.signed)) return { qualities: qualities, subs: subs };
            var map = {};
            signData.signed.forEach(function (s) { map[s.q] = s.src; });
            qualities.forEach(function (q) { if (map[q.q]) q.url = map[q.q]; });
            return { qualities: qualities, subs: subs };
        });
    }

    fetch(resolveUrl, { credentials: 'same-origin' })
        .then(function (r) { return r.ok ? r.json() : null; })
        .catch(function () { return null; })
        .then(function (resolveData) {
            if (!resolveData || !resolveData.id) {
                showError('Oynatıcı yüklenemedi (resolve hatası).');
                return null;
            }
            // Upstream API cagrisi sandbox iframe icinden - Origin: null gider.
            // Sandbox reject olursa (tau-video.xyz /api/video BOS govde donerse r.json()
            // patlar, ya da 15sn timeout) showError ile sonraki sunucuya OTOMATIK gec.
            // ESKIDEN burada sadece null donuyordu → `if (npData === null) return;` sessizce
            // cikiyor, calisan 2. embed (sibnet/dzen) denenmeden SIYAH EKRAN kaliyordu
            // (2026-05-30 tau API site-genel bos-200 regresyonu). Her iki null yolu da
            // (resolveData-fail + sandbox-fail) artik showError cagirir → tek geçiş.
            return czNpApiViaSandbox(resolveData.id).catch(function () { showError('Kaynak çözümlenemedi.'); return null; });
        })
        .then(function (npData) {
            if (npData === null) return; // resolve veya sandbox fail; showError ZATEN cagrildi (otomatik gecis yapildi)
            if (!npData) {
                showError('Oynatıcı API\'sine ulaşılamadı.');
                return;
            }
            if (!Array.isArray(npData.urls) || !npData.urls.length) {
                showError('Bu bölüm için kaynak bulunamadı.');
                return;
            }

            // urls[] -> {q, label, url} formatina cevir. Filtre = format kontrolu
            // ('<sayi>p'), beyaz liste DEGIL; siralama yukseklige gore azalan.
            var urlMap = {};
            npData.urls.forEach(function (u) {
                var q = String(u.label || '').toLowerCase();
                if (czHeightFromLabel(q) > 0 && u.url) urlMap[q] = String(u.url);
            });

            var qualities = Object.keys(urlMap)
                .sort(function (a, b) { return czHeightFromLabel(b) - czHeightFromLabel(a); })
                .map(function (q) { return { q: q, label: q.toUpperCase(), url: urlMap[q] }; });

            if (!qualities.length) {
                showError('Desteklenen kalite bulunamadı.');
                return;
            }

            // Caption discovery: sandbox iframe animecix episode-videos API'sini de
            // cagirdi ve eslesen embed_id'nin captions[] dizisini ekledi. Bos olabilir
            // (eski bolumlerde caption yok veya dub-only).
            var captions = Array.isArray(npData.captions) ? npData.captions : [];

            // URL'leri imzala (DOM'da upstream gozukmesin); captions[] de ayni POST'ta
            // /api/np-sub/?s=<sig> formatinda imzalanir. Fail olursa direct URL + bos sub.
            maybeSignQualities(qualities, captions).then(function (result) {
                var signedQualities = result.qualities;
                var signedSubs      = result.subs || [];

                // ArtPlayer subList format: {label, url, default}. Default = kullanici
                // tercih ettigi dil (czPlayerPrefs.subtitleLang), yoksa ilk siradaki.
                var subList = null;
                if (signedSubs.length) {
                    var prefLang = (window.czPlayerPrefs && window.czPlayerPrefs.subtitleLang)
                                 ? String(window.czPlayerPrefs.subtitleLang).toLowerCase() : 'tr';
                    var defIdx = -1;
                    for (var si = 0; si < signedSubs.length; si++) {
                        if (signedSubs[si].lang === prefLang) { defIdx = si; break; }
                    }
                    if (defIdx < 0) defIdx = 0;
                    subList = signedSubs.map(function (s, i) {
                        return { label: s.name || s.lang.toUpperCase(), url: s.src, default: i === defIdx };
                    });
                }

                var dataLike = { qualities: signedQualities, 'default': NP_DEFAULT };

                // 1) Manuel kullanici tercihi
                var pref = czGetUserQualityPref();
                if (pref && signedQualities.some(function (q) { return q.q === pref; })) {
                    czFinalizeDzenInit(el, dataLike, pref, false /* abrEnabled */,
                                       title, epLabel, nextUrl, malId, epNo, epId, subList);
                    return;
                }

                // 2) Bandwidth tahmin (probe yok - CORS engelli; connection API/default).
                czEstimateBandwidthNp().then(function (est) {
                    var picked = czPickQuality(signedQualities, est.mbps) || NP_DEFAULT || signedQualities[0].q;
                    if (window.console && console.info) {
                        console.info('[ABR/np] bandwidth=' + est.mbps.toFixed(2) + ' Mbps (' + est.source + ') → ' + picked);
                    }
                    czFinalizeDzenInit(el, dataLike, picked, true /* abrEnabled */,
                                       title, epLabel, nextUrl, malId, epNo, epId, subList);
                });
            });
        });
}

// Dzen + native opaque ABR akışında manifest + seçilen q ile player'i kurar.
// Tarihsel sebeple "DzenInit" - generic, birden fazla kaynak icin kullaniliyor.
// subList: tau np flow icin animecix episode-videos'tan gelen ASS altyazi listesi
// (signNpUrls signedSubs[] -> /api/np-sub/?s=<sig> proxy ile VTT'ye cevrilir).
// Dzen call-site'lardan subList gelmez -> default null, CC menusu olusmaz.
function czFinalizeDzenInit(el, data, chosenQ, abrEnabled, title, epLabel, nextUrl, malId, epNo, epId, subList) {
    subList = subList || null;
    var qArr = data.qualities.map(function (q) {
        return { html: q.label, url: q.url, default: q.q === chosenQ };
    });

    // 4K AI: WebGPU API mevcutsa kalite menusunun EN USTUNE "4K ⚡ AI" item'i ekle.
    // url = en yuksek kaynak + "&_a4k=1" extra query param'i. Server (npStream) sadece
    // s= imzasini dogrular, diger paramlari ignore eder. ArtPlayer'in URL dedupe
    // mantigi bu sekilde yeni URL'i farkli olarak gorur (fragment denendi, ArtPlayer
    // dedupe ediyordu → 'url' event hic fire etmiyordu, console bos kaliyordu).
    if (window.czAnime4kHasGpuApi && qArr.length) {
        var topQ = data.qualities[0]; // yukseklige gore azalan sirali: ilk = en yuksek
        if (topQ && topQ.url) {
            var sep = topQ.url.indexOf('?') >= 0 ? '&' : '?';
            qArr.unshift({
                html: '<span style="color:#fbbf24;font-weight:700">4K</span>',
                url: topQ.url + sep + '_a4k=1',
            });
            if (window.console) console.info('[4kai] menu item added, base url:', topQ.url);
        }
    } else if (window.console) {
        console.info('[4kai] WebGPU API yok (navigator.gpu undefined), 4K AI item eklenmedi - hasGpuApi:', window.czAnime4kHasGpuApi, ', qArr.length:', qArr.length);
    }

    var startEntry = data.qualities.find(function (q) { return q.q === chosenQ; })
                  || data.qualities[0];
    czBuildPlayer(el, startEntry.url, title, epLabel, nextUrl, malId, epNo, epId, qArr, abrEnabled, subList, null);
}

// HDFC HLS için manifest parse olduktan sonra kalite kontrolünü ekler.
// Audio control artık construct-time (czBuildPlayer içinde, audioList ile);
// burası sadece kalite varyantı (>1 BANDWIDTH) için çalışır.
// Kullanici default kalite tercihinden (settings_player.php) en yakin HLS level'i bulur.
// 'auto' veya esleme yoksa null doner -> Otomatik (currentLevel = -1) kalir.
function czPickPreferredLevel(levels) {
    var pref = window.czPlayerPrefs && window.czPlayerPrefs.quality;
    if (!pref || pref === 'auto' || !Array.isArray(levels) || !levels.length) return null;
    var target = parseInt(pref, 10);
    if (!isFinite(target) || target <= 0) return null;
    // En yakin level (height farki minimum). Esitlikte, target'in altinda kalan tercih edilir
    // (kullanici 720 istemis -> 1080'e yukselmek yerine 720'a uydurmak istiyoruz).
    var bestIdx = -1, bestDiff = Infinity;
    for (var i = 0; i < levels.length; i++) {
        var h = levels[i].height || 0;
        if (h <= 0) continue;
        var diff = Math.abs(h - target);
        if (diff < bestDiff || (diff === bestDiff && h <= target)) {
            bestDiff = diff; bestIdx = i;
        }
    }
    return bestIdx >= 0 ? bestIdx : null;
}

function czHdfcWireDynamicControls(art, hls) {
    // HLS MANIFEST_PARSED bazen iki kez fire olabiliyor (level switch sonrasi
    // re-emit). ArtPlayer ayni isimde kontrol icin "Can't add an existing" atar.
    // Flag ile ilk basarili eklemede mark, ikincide skip — hata console'a
    // sizmaz ve duplicate UI olmaz.
    if (art._czQualityControlAdded) return;
    try {
        var levels = hls.levels || [];
        if (levels.length > 1) {
            var preferredIdx = czPickPreferredLevel(levels);
            var qItems = [{ html: 'Otomatik', value: -1, default: preferredIdx === null }];
            var indexed = levels.map(function (l, i) { return { l: l, idx: i }; });
            indexed.sort(function (a, b) { return (b.l.height || 0) - (a.l.height || 0); });
            indexed.forEach(function (o) {
                qItems.push({
                    html: (o.l.height || '?') + 'p',
                    value: o.idx,
                    default: preferredIdx !== null && preferredIdx === o.idx,
                });
            });
            // Preferred level secildiyse hls'i de o seviyeye al (UI ile akis senkron olsun).
            if (preferredIdx !== null) {
                try { hls.currentLevel = preferredIdx; } catch (e) {}
            }
            art.controls.add({
                name: 'cz-quality',
                position: 'right',
                index: 37,
                html: preferredIdx !== null ? ((levels[preferredIdx].height || '?') + 'p') : 'Auto',
                style: { padding: '0 8px', fontSize: '12px', fontWeight: 'bold' },
                selector: qItems,
                onSelect: function (item) {
                    try { hls.currentLevel = item.value; } catch (e) {}
                    return item.html;
                },
            });
            art._czQualityControlAdded = true;
        }
    } catch (err) {
        if (window.console && console.warn) console.warn('[hdfc] dynamic controls failed', err);
    }
}

// Mevcut subList + audioList'ten hangi quick-mode butonlarının gözükeceğini hesapla.
// Her mod için: id (key), label (görünen metin), audioIdx (-1 = değişmez), subUrl ('' = kapalı),
// initial (true → ilk açılışta aktif).
function czHdfcResolveQuickModes(subList, audioList) {
    var modes = [];
    var trAudioIdx = -1, oriAudioIdx = -1;
    if (Array.isArray(audioList)) {
        for (var i = 0; i < audioList.length; i++) {
            var n = String(audioList[i].name || '').toLowerCase();
            if (trAudioIdx < 0 && (n.indexOf('turk') >= 0 || n.indexOf('türk') >= 0)) trAudioIdx = i;
            else if (oriAudioIdx < 0) oriAudioIdx = i;
        }
    }
    var trSubUrl = '';
    if (Array.isArray(subList)) {
        // "Forced" tipi sub'ı atla, sadece tam Türkçe altyazıyı tercih et.
        for (var j = 0; j < subList.length; j++) {
            var lbl = String(subList[j].label || '').toLowerCase();
            if (subList[j].lang === 'tr' && lbl.indexOf('forced') < 0) {
                trSubUrl = subList[j].url; break;
            }
        }
        // Hiç tam-TR yoksa Forced'a düş
        if (!trSubUrl) {
            for (var k = 0; k < subList.length; k++) {
                if (subList[k].lang === 'tr') { trSubUrl = subList[k].url; break; }
            }
        }
    }

    if (trAudioIdx >= 0) {
        modes.push({ id: 'dub', label: 'Türkçe Dublaj', audioIdx: trAudioIdx, subUrl: '' });
    }
    if (oriAudioIdx >= 0 && trSubUrl) {
        modes.push({ id: 'sub', label: 'Türkçe Altyazılı', audioIdx: oriAudioIdx, subUrl: trSubUrl });
    }
    if (oriAudioIdx >= 0) {
        modes.push({ id: 'org', label: 'Orijinal (Altyazısız)', audioIdx: oriAudioIdx, subUrl: '' });
    }

    // İlk açılış aktif modu seç:
    //   1. Once kullanici tercihi (settings_player.php)
    //      - audio=tr & sub varsa  -> Türkçe Dublaj (varsa) / Türkçe Altyazılı / Orijinal
    //      - audio=en & sub=tr     -> Türkçe Altyazılı (varsa) / Orijinal
    //      - audio=en & sub=en     -> Orijinal (TR sub'i agresif sok-mama; eldeki sub TR oldugu icin agresif applyMode'a girmesin)
    //   2. Tercih olmazsa eski varsayilan: TR audio default ise dub, yoksa sub varsa sub, yoksa son mod.
    var prefs = window.czPlayerPrefs || {};
    var prefAudio = prefs.audioLang === 'en' ? 'en' : 'tr';
    var prefSub   = prefs.subtitleLang === 'en' ? 'en' : 'tr';

    var defaultAudioIdx = -1;
    if (Array.isArray(audioList)) {
        for (var d = 0; d < audioList.length; d++) {
            if (audioList[d].default) { defaultAudioIdx = d; break; }
        }
    }

    var hasMode = function(id) { return modes.some(function(m){ return m.id === id; }); };
    var initialId = null;

    if (prefAudio === 'tr') {
        // Kullanici TR ses istiyor.
        if (hasMode('dub')) initialId = 'dub';
        else if (prefSub === 'tr' && hasMode('sub')) initialId = 'sub';
        else if (hasMode('org')) initialId = 'org';
    } else {
        // Kullanici EN/Orijinal ses istiyor.
        if (prefSub === 'tr' && hasMode('sub')) initialId = 'sub';
        else if (hasMode('org')) initialId = 'org';
        else if (hasMode('dub')) initialId = 'dub';
    }

    // Tercih hicbir mode'a uymadiysa eski default davranis:
    if (!initialId) {
        if (defaultAudioIdx === trAudioIdx && trAudioIdx >= 0) initialId = 'dub';
        else if (hasMode('sub')) initialId = 'sub';
        else if (modes.length) initialId = modes[modes.length - 1].id;
    }

    modes.forEach(function (m) { m.initial = (m.id === initialId); });
    return modes;
}

// Quick-mode butonuna tıklanınca: audio + subtitle state'ini değiştir.
function czHdfcApplyQuickMode(art, modeId, modes) {
    var m = null;
    for (var i = 0; i < modes.length; i++) if (modes[i].id === modeId) { m = modes[i]; break; }
    if (!m) return;
    try {
        if (art.hls && m.audioIdx >= 0 && art.hls.audioTracks && art.hls.audioTracks[m.audioIdx]) {
            art.hls.audioTrack = m.audioIdx;
        }
    } catch (e) {}
    try {
        if (m.subUrl) {
            art.subtitle.switch(m.subUrl, { name: m.label });
            art.subtitle.show = true;
        } else {
            art.subtitle.show = false;
        }
    } catch (e) {}

    // Aktif buton class'ı (layer DOM'unda)
    var root = (art.template && art.template.$player)
        ? art.template.$player.querySelector('[data-cz-qmode-root]')
        : document.querySelector('[data-cz-qmode-root]');
    if (root) {
        root.querySelectorAll('[data-qmode]').forEach(function (b) {
            b.classList.toggle('active', b.getAttribute('data-qmode') === modeId);
        });
    }

    // Audio + subtitle dropdown menülerinin label'larını da senkronize et
    czHdfcSyncQuickMode(art);
}

// Audio veya subtitle dropdown'ından değişiklik yapılınca, quick-mode butonlarının
// aktif state'ini eşleşen moda eşitle (ya da hiçbirine eşleşmiyorsa hepsini deaktive et).
function czHdfcSyncQuickMode(art) {
    var root = (art.template && art.template.$player)
        ? art.template.$player.querySelector('[data-cz-qmode-root]')
        : document.querySelector('[data-cz-qmode-root]');
    if (!root) return;

    var modes = art._czQuickModes || [];
    var curAudio = -1;
    try { if (art.hls) curAudio = art.hls.audioTrack; } catch (e) {}
    var subOn = false;
    try { subOn = !!(art.subtitle && art.subtitle.show); } catch (e) {}

    // State → mode eşleme: audio index + (mod altyazılı mı/değil mi). URL
    // karşılaştırması yapmıyoruz çünkü Artplayer'ın `subtitle.url` getter'ı
    // bazı sürümlerde absolutize / boş dönebiliyor → tutarsız eşleşme. Quick-mode
    // tanımında her audio+subState kombinasyonu için tek mod var, bu yeterli.
    var matchedId = null;
    for (var i = 0; i < modes.length; i++) {
        var m = modes[i];
        if (m.audioIdx !== curAudio) continue;
        var modeHasSub = !!m.subUrl;
        if (modeHasSub === subOn) { matchedId = m.id; break; }
    }

    var btns = root.querySelectorAll('[data-qmode]');
    btns.forEach(function (b) {
        b.classList.toggle('active', matchedId !== null && b.getAttribute('data-qmode') === matchedId);
    });
}

function czBuildPlayer(el, src, title, epLabel, nextUrl, malId, epNo, epId, qualityArr, abrEnabled, subList, audioList, youtubeId) {
    if (!el || typeof Artplayer === 'undefined') return null;
    subList   = subList   || null;
    audioList = audioList || null;
    youtubeId = youtubeId || '';
    var isYoutube = !!youtubeId;

    function escAttr(s) {
        return String(s).replace(/&/g,'&amp;').replace(/"/g,'&quot;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
    }

    // Hız seçenekleri 0.5x → 4x
    Artplayer.PLAYBACK_RATE = [0.5, 1, 1.5, 2, 2.5, 3, 3.5, 4];
    // Sağ-tık menüsündeki varsayılan öğeleri kaldır (Video Info vb.)
    Artplayer.CONTEXTMENU = [];

    var infoText = (title + (epLabel ? ' ' + epLabel : '')).trim();
    var epInfoHtml = infoText
        ? '<span class="cz-ep-info"><span class="cz-ep-dot">•</span><span>' + escAttr(infoText) + '</span></span>'
        : '';

    var customControls = [];
    if (epInfoHtml) {
        customControls.push({
            name: 'cz-ep-info',
            position: 'left',
            index: 40,
            html: epInfoHtml,
            click: function () {},
        });
    }

    // Hız selector - kalite menüsü gibi sağda görünür, tıklayınca dropdown açar.
    // ArtPlayer'in dahili `setting` panelini bu yüzden kaldırıyoruz; hız orada
    // tek seçenekti, yerine direkt buton koyuyoruz.
    var speedRates = [
        { html: '0.5x', value: 0.5 },
        { html: '1.0x', value: 1, default: true },
        { html: '1.25x', value: 1.25 },
        { html: '1.5x', value: 1.5 },
        { html: '2.0x', value: 2 },
        { html: '3.0x', value: 3 },
        { html: '4.0x', value: 4 },
    ];
    customControls.push({
        name: 'cz-speed',
        position: 'right',
        index: 38,
        html: '1.0x',
        style: { padding: '0 8px', fontSize: '13px', minWidth: '40px', textAlign: 'center' },
        selector: speedRates,
        onSelect: function (item) {
            try { this.playbackRate = item.value; } catch (e) {}
            return item.html;
        },
    });

    // HDFC ses dili picker - audioList >1 ise sağ tarafa selector ekle.
    // Construct-time'da ekleniyor (runtime add Artplayer'da bazen mount olmuyordu).
    // hls.js audio track index'i m3u8 sırasıyla aynı → audioList[i] hls.audioTracks[i].
    if (audioList && audioList.length > 1) {
        var aItems = audioList.map(function (a, i) {
            return { html: a.name || ('Ses ' + (i + 1)), value: i, default: !!a.default };
        });
        customControls.push({
            name: 'cz-audio',
            position: 'right',
            index: 35,
            html: '<i class="fas fa-language"></i>',
            style: { padding: '0 8px', fontSize: '13px' },
            selector: aItems,
            onSelect: function (item) {
                try { if (this.hls) this.hls.audioTrack = item.value; } catch (e) {}
                czHdfcSyncQuickMode(this);
                return '<i class="fas fa-language"></i> ' + item.html;
            },
        });
    }

    // HDFC subtitle picker - subList gelirse, sağ tarafa "Altyazı" selector ekle.
    // Default kalmazsa "Kapalı" başlangıç olur.
    if (subList && subList.length) {
        var subItems = [{ html: 'Kapalı', value: '', default: false }];
        var hasDefault = false;
        subList.forEach(function (s) {
            var isDef = !!s.default;
            if (isDef) hasDefault = true;
            subItems.push({ html: s.label, value: s.url, default: isDef });
        });
        if (!hasDefault) subItems[0].default = true;
        customControls.push({
            name: 'cz-sub',
            position: 'right',
            index: 36,
            html: 'CC',
            style: { padding: '0 8px', fontSize: '12px', fontWeight: 'bold', minWidth: '30px', textAlign: 'center' },
            selector: subItems,
            onSelect: function (item) {
                try {
                    if (!item.value) {
                        this.subtitle.show = false;
                    } else {
                        this.subtitle.switch(item.value, { name: item.html });
                        this.subtitle.show = true;
                    }
                } catch (e) {}
                czHdfcSyncQuickMode(this);
                return item.html === 'Kapalı' ? 'CC' : item.html;
            },
        });
    }

    // Quick-mode butonları için HTML hazırla - actual layer ekleme aşağıda
    // (new Artplayer'den sonra closure ile event handler bağlamak için).
    var quickModes = czHdfcResolveQuickModes(subList, audioList);
    var quickModesHtml = '';
    if (quickModes.length >= 2) {
        quickModesHtml = '<div class="cz-qmode" data-cz-qmode-root>';
        quickModes.forEach(function (m) {
            quickModesHtml += '<button type="button" data-qmode="' + m.id + '"'
                            + (m.initial ? ' class="active"' : '') + '>'
                            + m.label + '</button>';
        });
        quickModesHtml += '</div>';
    }

    var layers = [];
    if (nextUrl) {
        layers.push({
            name: 'cz-next',
            html:
                // DIKKAT: <a href> DEGIL. Reklam scripti (main.php'deki wargamings)
                // sayfadaki her <a href> uzerine gorunmez, position:fixed,
                // z-index:2147483647, pointer-events:auto bir tik-yakalayici div
                // koyuyor. Rozet videonun USTUNDE durdugu icin bu yakalayici,
                // rozet daha gorunmezken (pointer-events:none) bile videonun
                // sag-alt bolgesini kapliyor ve tiki alttaki linke yonlendiriyordu
                // -> kullanici bolumun HERHANGI bir aninda videoya tiklayinca
                // sonraki bolume atiliyordu (2026-08-06). Link olmayan bir
                // <button> reklam scripti tarafindan indekslenmiyor.
                '<button type="button" class="cz-next-ep" data-next-url="' + escAttr(nextUrl) + '">' +
                    '<svg viewBox="0 0 24 24" fill="currentColor"><path d="M6 4l12 8-12 8V4zm13 0h2v16h-2V4z"/></svg>' +
                    'Sıradaki Bölüm' +
                '</button>',
        });
    }

                    // video element bos buffer'la stall olur, video:error fire eder,
                    // kullaniciya "Video yuklenemedi" gozukur (console temiz cunku
                    // hls.js DEFAULT'ta sessizdir). Asagidaki handler:
                    //   - NETWORK_ERROR (frag/level/manifest) -> startLoad() ile retry
                    //   - MEDIA_ERROR -> recoverMediaError() (codec/buffer sorunu)
                    //   - Recover sayisi 3'u gecerse fallback'e birak (next server)
                    var hdfcRecover = { net: 0, media: 0 };

                        if (window.console && console.warn) {

                                data.url || '');

    // Tap-toggle ikonu icin .cz-playing + buffering icin .cz-loading wrapper

    // degil - kendi class'imizi koyuyoruz.
    var $czWrap = (art.template && art.template.$player) ? art.template.$player : el;

    // Double-tap detector - sag yari +10s, sol yari -10s. Center +/- %18'lik
    // bolge merkez pause/play butonuna ayrildi (double-tap absorb etmez).
    if ($czWrap) {
        var DT_MS = 320;
        var DT_PX = 50;
        var CENTER_BAND = 0.18;
        var lastTap = 0;
        var lastX = 0;
        var bubbleTimer = null;
        // Mobilde her tap icin once 'touchend' sonra ~300ms gecikmeli 'click'
        // tetikleniyor → tek dokunus iki kez handler'a giriyor ve double-tap
        // sayiliyordu. touchend sonrasi short window'da click bastir. */
        var suppressClickUntil = 0;

        function flashBubble(side) {
            var sel = side === 'right'
                ? '.cz-seek-bubble.cz-seek-right'
                : '.cz-seek-bubble.cz-seek-left';
            var $b = $czWrap.querySelector(sel);
            if (!$b) return;
            $b.classList.add('show');
            if (bubbleTimer) clearTimeout(bubbleTimer);
            bubbleTimer = setTimeout(function () {
                $b.classList.remove('show');
            }, 600);
        }

        function doSeek(delta) {
            try {
                var dur = art.duration || 0;
                var nt = (art.currentTime || 0) + delta;
                if (nt < 0) nt = 0;
                if (dur && nt > dur - 0.5) nt = Math.max(0, dur - 0.5);
                art.seek = nt;
            } catch (e) {}
        }

        function onTap(e) {
            var t = e.target;
            if (t && t.closest && (
                t.closest('.art-bottom') ||
                t.closest('.art-controls') ||
                t.closest('.cz-mobile-toggle') ||
                t.closest('.cz-qmode') ||
                t.closest('.cz-next-ep')
            )) return;

            var touch = (e.changedTouches && e.changedTouches[0]) || e;
            if (typeof touch.clientX !== 'number') return;

            var rect = $czWrap.getBoundingClientRect();
            var w = rect.width;
            if (!w) return;
            var x = touch.clientX - rect.left;
            var fr = x / w;

            if (fr > 0.5 - CENTER_BAND && fr < 0.5 + CENTER_BAND) {
                lastTap = 0;
                return;
            }

            var now = Date.now();
            var isDouble = (now - lastTap < DT_MS) && (Math.abs(x - lastX) < DT_PX);

            if (isDouble) {
                e.preventDefault && e.preventDefault();
                var isRight = fr > 0.5;
                doSeek(isRight ? 10 : -10);
                flashBubble(isRight ? 'right' : 'left');
                lastTap = 0;
            } else {
                lastTap = now;
                lastX = x;
            }
        }

            suppressClickUntil = Date.now() + 450;
            onTap(e);
        }, { passive: false });
        $czWrap.addEventListener('click', function (e) {
            if (Date.now() < suppressClickUntil) return;
            onTap(e);
        });

        /* ===== Hold-to-2x: mouse uzun-bas / touch uzun-bas / space basili tut
           500ms esikten sonra playbackRate=2; birakinca eski hiza doner.
           Trailing click suppress edilir ki tap-to-toggle yanlislikla tetiklenmesin.
           Spacebar icin ArtPlayer'in dahili hotkey'i capture-phase'de iptal edilir;
           kisa basista art.toggle() manuel cagrilir. */
        (function () {
            var HOLD_MS = 500;
            var holdTimer = null;
            var holding = false;
            var savedRate = 1;
            var pointerDown = false;
            var spaceDown = false;
            var $bubble = $czWrap.querySelector('.cz-speed-bubble');

            function inControl(t) {
                return !!(t && t.closest && (
                    t.closest('.art-bottom') ||
                    t.closest('.art-controls') ||
                    t.closest('.art-control') ||
                    t.closest('.art-settings') ||
                    t.closest('.art-info') ||
                    t.closest('.cz-mobile-toggle') ||
                    t.closest('.cz-qmode') ||
                    t.closest('.cz-next-ep')

            }

            function startHold() {
                if (holding) return;
                try { savedRate = art.playbackRate || 1; } catch (e) { savedRate = 1; }
                holding = true;
                try { art.playbackRate = 2; } catch (e) {}
                if ($bubble) $bubble.classList.add('show');
            }

            function endHold(swallowClick) {
                if (holdTimer) { clearTimeout(holdTimer); holdTimer = null; }
                if (!holding) return;
                holding = false;
                try { art.playbackRate = savedRate || 1; } catch (e) {}
                if ($bubble) $bubble.classList.remove('show');
                if (swallowClick) {
                    suppressClickUntil = Date.now() + 500;
                    var oneShot = function (ev) {
                        ev.stopPropagation();
                        ev.stopImmediatePropagation();
                        ev.preventDefault();
                        $czWrap.removeEventListener('click', oneShot, true);
                    };
                    $czWrap.addEventListener('click', oneShot, true);
                }
            }

            // --- Mouse hold (yalniz sol tus, kontrol bar disinda) ---
            $czWrap.addEventListener('mousedown', function (e) {
                if (e.button !== 0) return;
                if (inControl(e.target)) return;
                pointerDown = true;
                if (holdTimer) clearTimeout(holdTimer);
                holdTimer = setTimeout(startHold, HOLD_MS);
            });
            window.addEventListener('mouseup', function () {
                if (!pointerDown) return;
                pointerDown = false;
                var wasHolding = holding;
                if (holdTimer) { clearTimeout(holdTimer); holdTimer = null; }
                if (wasHolding) endHold(true);
            });

            // --- Touch hold (tek parmak; multi-touch ve drag iptal eder) ---
            $czWrap.addEventListener('touchstart', function (e) {
                if (e.touches.length !== 1) {
                    if (holdTimer) { clearTimeout(holdTimer); holdTimer = null; }
                    if (holding) endHold(true);
                    return;
                }
                if (inControl(e.target)) return;
                if (holdTimer) clearTimeout(holdTimer);
                holdTimer = setTimeout(startHold, HOLD_MS);
            }, { passive: true });
            $czWrap.addEventListener('touchmove', function () {
                if (holdTimer && !holding) {
                    clearTimeout(holdTimer);
                    holdTimer = null;
                }
            }, { passive: true });
            var endTouch = function () {
                var wasHolding = holding;
                if (holdTimer) { clearTimeout(holdTimer); holdTimer = null; }
                if (wasHolding) endHold(true);
            };
            $czWrap.addEventListener('touchend', endTouch, { passive: true });
            $czWrap.addEventListener('touchcancel', endTouch, { passive: true });

            // --- Spacebar: capture-phase'de ArtPlayer'in hotkey'ini bastir.
            // Kisa basista art.toggle() (esdeger davranis); 500ms+ tutuldugunda 2x. */
            document.addEventListener('keydown', function (e) {
                if (e.code !== 'Space' && e.key !== ' ') return;
                if (!window.czArtPlayer || window.czArtPlayer !== art) return;
                var ae = document.activeElement;
                if (ae && (ae.tagName === 'INPUT' || ae.tagName === 'TEXTAREA' || ae.isContentEditable)) return;
                e.preventDefault();
                e.stopImmediatePropagation();
                if (e.repeat) return;
                if (spaceDown) return;
                spaceDown = true;
                if (holdTimer) clearTimeout(holdTimer);
                holdTimer = setTimeout(startHold, HOLD_MS);
            }, true);
            document.addEventListener('keyup', function (e) {
                if (e.code !== 'Space' && e.key !== ' ') return;
                if (!spaceDown) return;
                spaceDown = false;
                var wasHolding = holding;
                if (holdTimer) { clearTimeout(holdTimer); holdTimer = null; }
                if (wasHolding) {
                    endHold(false);
                } else {
                    try { art.toggle(); } catch (err) {}
                }
            }, true);

            // Sayfa visibility kaybi / blur'da takili kalmayi onle
            window.addEventListener('blur', function () {
                pointerDown = false;
                spaceDown = false;
                if (holdTimer) { clearTimeout(holdTimer); holdTimer = null; }
                if (holding) endHold(false);
            });
        })();
    }

    if ($czWrap) {
        // Init: ilk acilista da spinner gozukurken merkez play butonu arkada
        // kalmasin → cz-loading'i basta ekle, ilk canplay/loadeddata'da kaldir.
        // cz-first-load: "ilk acilista uzun surebilir" bilgi metnini sadece
        // gercekten ilk yuklemede goster; ilk video:playing'de kaldir.
        $czWrap.classList.add('cz-loading');
        if (!window.__czFirstLoadDone) $czWrap.classList.add('cz-first-load');

        art.on('play', function () {
            window.__czFirstLoadDone = true;
            $czWrap.classList.add('cz-playing');
            $czWrap.classList.remove('cz-loading');
            $czWrap.classList.remove('cz-first-load');
        });
        art.on('video:playing', function () {
            window.__czFirstLoadDone = true;
            $czWrap.classList.add('cz-playing');
            $czWrap.classList.remove('cz-loading');
            $czWrap.classList.remove('cz-first-load');
        });
        art.on('pause', function () {
            $czWrap.classList.remove('cz-playing');
            $czWrap.classList.remove('cz-loading');
        });
        art.on('video:loadstart', function () {
            $czWrap.classList.add('cz-loading');
        });
        art.on('video:waiting', function () {
            $czWrap.classList.add('cz-loading');
        });
        art.on('video:canplay', function () {
            $czWrap.classList.remove('cz-loading');
        });
        art.on('video:loadeddata', function () {
            $czWrap.classList.remove('cz-loading');
        });

        // video:error — stream yuklenemezse sonraki denenmemis server'a gec.
        // Auto-fallback mantigi czHandleServerFailure'da merkezilesti:
        // (1) ayni dilde sonraki denenmemis embed, (2) diger dil havuzu,
        // (3) hicbiri kalmadiysa retry overlay. Iframe load timeout'u da ayni
        // entry-point'i kullanir - tek dogru kaynak.
        var _errFired = false;
        art.on('video:error', function () {
            if (_errFired) return;
            _errFired = true;
            try { art.destroy(); } catch (e) {}
            if (typeof window.czHandleServerFailure === 'function') {
                window.czHandleServerFailure();
            }
        });
    }

    // Quick-mode layer - Artplayer construct'ten sonra ekleniyor ki closure'da
    // `art` instance'ına erişip click handler'da doğrudan kullanabilelim.
    if (quickModesHtml) {
        art.layers.add({
            name: 'cz-qmode-layer',
            html: quickModesHtml,
            mounted: function ($el) {
                if (!$el || $el.dataset.czQmodeBound === '1') return;
                $el.dataset.czQmodeBound = '1';
                $el.addEventListener('click', function (e) {
                    var btn = e.target && e.target.closest && e.target.closest('[data-qmode]');
                    if (!btn) return;
                    czHdfcApplyQuickMode(art, btn.getAttribute('data-qmode'), quickModes);
                });
            },
        });
    }

    // Sıradaki Bölüm overlay'ini son 25 saniyede göster
    if (nextUrl) {
        art.on('video:timeupdate', function () {
            var dur = art.duration || 0;
            if (!dur || !isFinite(dur)) return;
            var btn = el.querySelector('.cz-next-ep');
            if (!btn) return;
            if (dur - art.currentTime <= 60) {
                btn.classList.add('show');
            } else {
                btn.classList.remove('show');
            }
        });

        // Oto Sonraki Bölüm: video bittiğinde, toggle açıksa sonraki bölüme git
        var autoNextFired = false;
        var czPlayerBuiltAt = Date.now();
        art.on('video:ended', function () {
            if (autoNextFired) return;
            // Gate: butonun görünen durumu belirleyici (bkz. window.czAutoNextOn).
            // Helper yoksa (script hatası) güvenli taraf: otomatik geçiş YOK.
            if (typeof window.czAutoNextOn !== 'function' || !window.czAutoNextOn()) return;
            // "Kaldığın yerden devam" bölümün sonuna yakın bir yere sardıysa,
            // sayfa açılır açılmaz 'ended' gelip kullanıcıyı sonraki bölüme
            // fırlatıyordu. En az 20sn oynatıcı ömrü şartı bunu engeller.
            if (Date.now() - czPlayerBuiltAt < 20000) return;
            autoNextFired = true;
            // Room mode + host: watch-party.js'in change_episode akışı kullanılır,
            // misafirler de yeni bölüme yönlenir. Diğer durumda normal navigasyon.
            if (typeof window.czRoomChangeEpisode === 'function'
                && window.czRoomChangeEpisode(nextUrl)) {
                return;
            }
            if (typeof window.czShowToast === 'function') {
                window.czShowToast('Sonraki bölüme geçiliyor…', 'success');
            }
            // Tam ekranda ise sayfayı yenilemeden geç (FS korunur); değilse normal git.
            setTimeout(function () {
                if (typeof window.czGoToNextEpisode === 'function') {
                    window.czGoToNextEpisode(nextUrl);
                } else {
                    window.location.href = nextUrl;
                }
            }, 600);
        });
    }

    // Oto Oynat: toggle açıksa video'yu otomatik başlat.
    // Tarayıcılar sesli autoplay'i çoğu zaman engeller - engellenirse muted olarak tekrar dene.
    try {
        if (localStorage.getItem('cz-autoplay') === '1') {
            var startAutoplay = function () {
                var p;
                try { p = art.play(); } catch (e) { return; }
                if (p && typeof p.then === 'function') {
                    p.catch(function () {
                        try {
                            art.muted = true;
                            var p2 = art.play();
                            if (p2 && typeof p2.catch === 'function') p2.catch(function(){});
                        } catch (e) {}
                    });
                }
            };
            // Player hazır olunca başlat (ready bazen kaçırılırsa fallback)
            art.on('ready', startAutoplay);
            setTimeout(startAutoplay, 700);
        }
    } catch (e) {}

    /* ===== Skip-time'lar: 3 kaynak paralel cekilir =====
       1) Manuel (kendi DB) - admin'in elle girdigi timestamp'ler
       2) AniSkip (canli) - anime icin (malId varsa)
       3) IntroDB (cache+live) - dizi icin (backend kind kontrolu yapar)
       Oncelik: manuel > AniSkip > IntroDB. Ayni skip_type'da manuel varsa ustteki kaynaklar ezilir. */
    console.log('[skip] init:', { malId: malId, epNo: epNo, epId: epId });

    var manualPromise = epId
        ? fetch('/api/skip-times/' + epId + '/', { credentials: 'same-origin' })
            .then(function (r) { return r.ok ? r.json() : { results: [] }; })
            .catch(function () { return { results: [] }; })
        : Promise.resolve({ results: [] });

    var aniskipPromise = (malId && epNo)
        ? fetch('https://api.aniskip.com/v2/skip-times/' + malId + '/' + epNo +
                '?types[]=op&types[]=ed&types[]=recap' +
                '&types[]=mixed-op&types[]=mixed-ed&episodeLength=0')
            .then(function (r) { return r.ok ? r.json() : { results: [] }; })
            .catch(function () { return { results: [] }; })
        : Promise.resolve({ results: [] });

    var introdbPromise = epId
        ? fetch('/api/introdb-skip/' + epId + '/', { credentials: 'same-origin' })
            .then(function (r) { return r.ok ? r.json() : { results: [] }; })
            .catch(function () { return { results: [] }; })
        : Promise.resolve({ results: [] });

    Promise.all([manualPromise, aniskipPromise, introdbPromise]).then(function (sources) {
        var manual  = sources[0] || { results: [] };
        var aniskip = sources[1] || { results: [] };
        var introdb = sources[2] || { results: [] };
        var results = [];
        var seenTypes = {};
        // 1) Manuel - en yuksek oncelik
        (manual.results || []).forEach(function (r) {
            results.push(r);
            seenTypes[r.skipType] = true;
        });
        // 2) AniSkip - manuel ile cakisanlari atla
        if (aniskip && aniskip.found && aniskip.results) {
            aniskip.results.forEach(function (r) {
                if (!seenTypes[r.skipType]) { results.push(r); seenTypes[r.skipType] = true; }
            });
        }
        // 3) IntroDB - manuel/aniskip ile cakisanlari atla
        if (introdb && introdb.results) {
            introdb.results.forEach(function (r) {
                if (!seenTypes[r.skipType]) { results.push(r); seenTypes[r.skipType] = true; }
            });
        }
        console.log('[skip] merged - manual:', (manual.results || []).length,
                    ', aniskip:', (aniskip.results || []).length,
                    ', introdb:', (introdb.results || []).length,
                    ', final:', results.length);
        if (results.length) czApplySkipTimes(art, el, results);
    });

    /* ===== Kaldığın yerden devam et (localStorage, episode başına) ===== */
    if (epId) {
        var resumeKey       = 'cz-resume-' + epId;
        var RESUME_MIN      = 10;     // <10sn kaydetme (yeni başlamış)
        // Bitime <60sn ise tamamlandı say. 30sn iken son dakikaya sarılan konum
        // kaydedilip geri yükleniyordu → bölüm açılır açılmaz bitiyor, kullanıcı
        // sonraki bölüme atılıyordu. Sınır "Sıradaki Bölüm" overlay'i (60sn) ile aynı.
        var RESUME_END_GAP  = 60;
        var RESUME_THROTTLE = 5000;   // ms - en sık 5sn'de bir yaz
        var lastSaveAt = 0;

        function czCanSave() {
            var t = art.currentTime;
            var d = art.duration;
            if (!isFinite(t) || t < RESUME_MIN) return null;
            if (d && isFinite(d) && t > d - RESUME_END_GAP) return null;
            return t;
        }

        function czSaveResume() {
            var t = czCanSave();
            if (t === null) return;
            try { localStorage.setItem(resumeKey, String(t)); } catch (e) {}
        }

        // Restore - metadata yüklenince duration güvenilir olur
        var restored = false;
        function czRestore() {
            if (restored) return;
            try {
                var saved = parseFloat(localStorage.getItem(resumeKey) || '0');
                if (!saved || saved < RESUME_MIN) { restored = true; return; }
                var d = art.duration;
                // Sure henuz bilinmiyorsa (ready, loadedmetadata'dan once gelebilir)
                // seek ETME - yoksa bitime yakin konum korumasiz geri yuklenir,
                // video aninda biter ve oto-sonraki kullaniciyi bolumden atar.
                // restored'i isaretlemeden cik: loadedmetadata tekrar cagiracak.
                if (!d || !isFinite(d)) return;
                restored = true;
                if (saved > d - RESUME_END_GAP) {
                    localStorage.removeItem(resumeKey);
                    return;
                }
                art.seek = saved;
                if (typeof window.czShowToast === 'function') {
                    var m = Math.floor(saved / 60);
                    var s = Math.floor(saved % 60);
                    var label = m + ':' + (s < 10 ? '0' + s : s);
                    window.czShowToast(label + ' konumundan devam ediliyor', 'info');
                }
            } catch (e) {}
        }
        art.on('video:loadedmetadata', czRestore);
        art.on('ready', czRestore);

        // Throttled save
        art.on('video:timeupdate', function () {
            var now = Date.now();
            if (now - lastSaveAt < RESUME_THROTTLE) return;
            lastSaveAt = now;
            czSaveResume();
        });

        // Pause / sayfa kapanma anında garanti kaydet
        art.on('video:pause', czSaveResume);
        window.addEventListener('pagehide', czSaveResume);
        window.addEventListener('beforeunload', czSaveResume);

        // Bölüm bitti -> kaydı sil
        art.on('video:ended', function () {
            try { localStorage.removeItem(resumeKey); } catch (e) {}
        });
    }

    /* ===== Adaptif kalite hook'lari (Dzen, abrEnabled=true iken) ===== */
    if (qualityArr && qualityArr.length) {
        // ArtPlayer kalite menusu kullaniciyla art.url'i degistirir. Biz manuel
        // (user click) vs otomatik (downgrade) ayrimini bir flag'le yapariz.
        art._czAutoChange = false;
        var stallTimes = [];
        var seekActiveAt = 0;
        var downgradedOnce = false;

        // 4K AI revert için: gözüktüğü her güvenli (non-a4k) URL'i sakla.
        // Initial: art.option.url ilk yüklenen URL — restart fire olmadan önce de geçerli.
        var lastSafeUrl = (art.option && art.option.url) || '';

        // ── 4K AI Premium Gate (DOM-level capture intercept) ──────────────────
        // ArtPlayer'in URL setter'i bazen aynı base URL icin restart fire etmiyor →
        // revert mekanizmasi guvenilmez. Daha saglam: quality menu item DOM'una
        // capture-phase click listener bagla, premium degilse ArtPlayer'in kendi
        // click handler'i hic calismadan e.stopPropagation() ile durdur + notice.
        (function () {
            var root = art.template && art.template.$player;
            if (!root) return;

            // Player içi custom notice — hiçbir ArtPlayer API'sine baglı degil, garanti calisir.
            function showA4kNotice() {
                var prev = root.querySelector('.cz-a4k-notice');
                if (prev) prev.remove();
                var n = document.createElement('a');
                n.className = 'cz-a4k-notice';
                n.href = '/premium/';
                n.innerHTML = '<span style="color:#fbbf24;margin-right:4px">⭐</span> 4K AI sadece <strong style="color:#fbbf24;text-decoration:underline">Premium</strong> üyelere özel';
                n.style.cssText = [
                    'position:absolute',
                    'top:50%','left:50%',
                    'transform:translate(-50%,-50%)',
                    'background:rgba(0,0,0,.9)',
                    'color:#fff',
                    'padding:14px 22px',
                    'border:1px solid rgba(251,191,36,.5)',
                    'border-radius:10px',
                    'font:600 14px/1.3 system-ui,sans-serif',
                    'box-shadow:0 6px 24px rgba(0,0,0,.6)',
                    'z-index:9999',
                    'cursor:pointer',
                    'text-decoration:none',
                    'display:inline-block',
                    'transition:transform .15s, box-shadow .15s',
                    'animation:czA4kNoticeIn .22s ease-out, czA4kNoticeOut .35s ease-in 2.7s forwards',
                ].join(';');
                n.addEventListener('mouseenter', function () {
                    n.style.transform = 'translate(-50%,-50%) scale(1.04)';
                    n.style.boxShadow = '0 8px 32px rgba(251,191,36,.35)';
                });
                n.addEventListener('mouseleave', function () {
                    n.style.transform = 'translate(-50%,-50%) scale(1)';
                    n.style.boxShadow = '0 6px 24px rgba(0,0,0,.6)';
                });
                if (!document.getElementById('cz-a4k-notice-css')) {
                    var st = document.createElement('style');
                    st.id = 'cz-a4k-notice-css';
                    st.textContent = '@keyframes czA4kNoticeIn{from{opacity:0;transform:translate(-50%,-50%) scale(.92)}to{opacity:1;transform:translate(-50%,-50%) scale(1)}}@keyframes czA4kNoticeOut{from{opacity:1}to{opacity:0;transform:translate(-50%,-50%) scale(.96)}}';
                    document.head.appendChild(st);
                }
                root.appendChild(n);
                setTimeout(function () { try { n.remove(); } catch (_) {} }, 3100);
            }

            var bindA4kItems = function () {
                // SADECE leaf menu item'lar: text content'i tam olarak "4K ⚡ AI" olan node'lar.
                // Parent container'lar (kalite cog button, settings panel) genis text icerir,
                // exact match onlari elemine eder.
                var nodes = root.querySelectorAll('.art-setting-item, .art-selector-item');
                for (var i = 0; i < nodes.length; i++) {
                    var node = nodes[i];
                    if (node._a4kBound) continue;
                    var leafText = (node.textContent || '').trim();
                    // ArtPlayer item.html: '<span...>4K</span>' → textContent "4K"
                    if (leafText !== '4K') continue;
                    node._a4kBound = true;
                    node.addEventListener('click', function (e) {
                        if (window.czAnime4kIsPremium) return; // premium → normal flow
                        e.stopPropagation();
                        e.stopImmediatePropagation && e.stopImmediatePropagation();
                        e.preventDefault();
                        showA4kNotice();
                    }, true /* capture */);
                }
            };
            // Quality menu lazy render edilir; MutationObserver ile yeni eklenen
            // item'lari yakala. Setup baslangic anında zaten varsa onlari da bagla.
            bindA4kItems();
            var observer = new MutationObserver(bindA4kItems);
            observer.observe(root, { childList: true, subtree: true });
        })();

        function urlToQ(u) {
            var m = String(u).match(/[?&]q=([a-z]+)/);
            return m ? m[1] : '';
        }

        // Manuel override: user kalite menusunden tikladi → pref kaydet, ABR durdur.
        // "_a4k=1" query param'li URL ozel: 4K AI overlay'i toggle eder, q-pref'e
        // dokunmaz (4K AI yapay bir kalite, kullanicinin gercek pref'i bozulmasin).
        //
        // NOT (2026-05-22): Eskiden art.on('url') kullaniliyordu ama ArtPlayer v5'te
        // boyle bir event YOK (kaynak: urlMix.js sadece 'restart' emit eder).
        // Eski ABR auto-downgrade ve manuel pref kaydetme path'leri olu koddu,
        // ABR dusurme stallTimes 3 dolduktan sonra 'restart' uzerinden fire eder.
        art.on('restart', function (newUrl) {
            if (window.console) console.info('[4kai] art.on(restart) fired:', String(newUrl).slice(0, 120));
            if (art._czAutoChange) { art._czAutoChange = false; return; }

            if (/[?&]_a4k=1(\b|$)/.test(String(newUrl))) {
                if (window.console) console.info('[4kai] 4K AI URL detected');

                // Premium gate: non-premium kullanicilar VE misafirler engellenir
                if (!window.czAnime4kIsPremium) {
                    // ArtPlayer notice — player içinde 3sn'lik küçük yazı, modal yok
                    try { art.notice.show = '⭐ 4K AI Premium üyelere özel'; } catch (_) {
                        if (window.czShowToast) window.czShowToast('⭐ 4K AI Premium üyelere özel', 'warning');
                    }
                    // URL'i son güvenli (non-a4k) URL'e geri çevir — _czAutoChange flag'i
                    // ikincil restart event'ini bypass eder, sonsuz döngü olmaz.
                    if (lastSafeUrl) {
                        art._czAutoChange = true;
                        try { art.url = lastSafeUrl; } catch (_) { art._czAutoChange = false; }
                    }
                    return;
                }

                // Mode resolve: login user → DB pref, misafir → localStorage fallback (default auto)
                var userMode = window.czAnime4kUserMode;
                if (!userMode) {
                    try { userMode = localStorage.getItem('cz_a4k_mode') || 'auto'; }
                    catch (_) { userMode = 'auto'; }
                }
                if (window.console) console.info('[4kai] enabling overlay, mode:', userMode);
                if (typeof window.czAnime4kEnable === 'function') {
                    window.czAnime4kEnable(art, userMode);
                }
                abrEnabled = false; // 4K AI'da ABR auto-downgrade kapali
                return;
            }

            // Güvenli (non-a4k) URL — revert için sakla
            lastSafeUrl = String(newUrl);
            // 4K AI dısı bir kalite secildi - overlay aktifse kapat
            if (typeof window.czAnime4kDisable === 'function') {
                window.czAnime4kDisable();
            }

            var q = urlToQ(newUrl);
            if (!q) return;
            czSetUserQualityPref(q);
            abrEnabled = false; // bundan sonra otomatik dusurme yapma
        });

        // Stall guard: gercek stall'lari say (seek-related haric).
        if (abrEnabled) {
            art.on('video:seeking', function () { seekActiveAt = Date.now(); });

            art.on('video:waiting', function () {
                if (downgradedOnce) return;
                var now = Date.now();
                if (now - seekActiveAt < 5000) return; // seek lag'i - say sayma

                // 30 sn pencere - disindakileri at
                stallTimes = stallTimes.filter(function (t) { return now - t < 30000; });
                stallTimes.push(now);

                if (stallTimes.length >= 3) {
                    // Bir alt kaliteye dus
                    var curUrl = art.url || '';
                    var curQ = urlToQ(curUrl);
                    if (!curQ) return;

                    var idx = qualityArr.findIndex(function (q) { return urlToQ(q.url) === curQ; });
                    if (idx < 0 || idx + 1 >= qualityArr.length) return; // zaten en dusuk

                    var nextEntry = qualityArr[idx + 1];
                    art._czAutoChange = true;
                    art.url = nextEntry.url; // ArtPlayer currentTime'i koruyup yeni kalitede devam eder
                    downgradedOnce = true;
                    stallTimes = [];

                    if (typeof window.czShowToast === 'function') {
                        window.czShowToast('Bağlantı zayıf - kalite ' + nextEntry.html + ' yapıldı.', 'info');
                    }
                    if (window.console) console.info('[ABR] auto-downgrade → ' + nextEntry.html);
                }
            });
        }
    }

    window.czPlayer = art;

    // Tam ekranda bölüm geçişinde FS geçici olarak .player-wrap'e taşınmıştı; yeni
    // player hazır olunca FS'i tekrar bu player'a ver. Belge zaten fullscreen'de
    // (wrap üzerinde) olduğu için bu transfer kullanıcı jesti gerektirmez; başarısız
    // olursa wrap-FS'te kalır (yine tam ekran, yalnızca ikon durumu farklı).
    if (window.czPendingFsRestore) {
        window.czPendingFsRestore = false;
        art.on('ready', function () {
            try { art.fullscreen = true; } catch (e) {}
        });
    }

    try {
        window.dispatchEvent(new CustomEvent('cz-player-ready', { detail: { art: art } }));
    } catch (e) {}
    return art;
}

/* AniSkip sonuçlarını ArtPlayer'a uygula:
   - progress bar üstüne işaret koy (highlight)
   - cz-skip-intro toggle açıksa op/recap segmentlerinde otomatik atla
   - kullanıcıya "Opening Atla" butonu göster (segment içindeyken) */
function czApplySkipTimes(art, container, results) {
    console.log('[aniskip] applying', results.length, 'skip segments');
    if (!art || !container || !results || !results.length) return;

    var TYPE_LABELS = {
        'op':        'Opening',
        'ed':        'Ending',
        'recap':     'Önceki Bölüm Özeti',
        'mixed-op':  'Opening',
        'mixed-ed':  'Ending'
    };

    // 1) Progress bar üzerine renkli bant overlay (op=sarı, ed=turuncu, recap=kırmızı)
    var BAND_COLORS = {
        'op':        'rgba(241, 196, 15, 0.75)',   // sarı
        'mixed-op':  'rgba(241, 196, 15, 0.75)',
        'ed':        'rgba(230, 126, 34, 0.75)',   // turuncu
        'mixed-ed':  'rgba(230, 126, 34, 0.75)',
        'recap':     'rgba(231, 76, 60, 0.75)'     // kırmızı
    };
    function renderSkipBands() {
        var progressEl = container.querySelector('.art-control-progress')
                      || container.querySelector('.art-progress')
                      || container.querySelector('[class*="progress"]');
        if (!progressEl) {
            console.log('[aniskip] progress bar element bulunamadi');
            return;
        }
        // önce eski bantları sil (re-render için)
        var olds = progressEl.querySelectorAll('.cz-skip-band');
        for (var i = 0; i < olds.length; i++) olds[i].remove();
        var dur = art.duration;
        if (!dur || !isFinite(dur) || dur <= 0) return;
        results.forEach(function (s) {
            if (!s.interval) return;
            var band = document.createElement('div');
            band.className = 'cz-skip-band';
            band.title = TYPE_LABELS[s.skipType] || s.skipType;
            var leftPct  = (s.interval.startTime / dur) * 100;
            var widthPct = ((s.interval.endTime - s.interval.startTime) / dur) * 100;
            band.style.cssText =
                'position:absolute;top:0;height:100%;' +
                'left:' + leftPct + '%;width:' + widthPct + '%;' +
